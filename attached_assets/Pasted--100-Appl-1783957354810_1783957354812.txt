package com.workshoptech.service

import android.Manifest

import android.content.Intent

import android.location.Location

import android.os.Looper
import androidx.core.app.NotificationCompat
import androidx.core.content.ContextCompat

import com.workshoptech.WorkshopTechApp

class LocationService : Service() {

    private lateinit var fusedLocationClient: FusedLocationProviderClient
    private lateinit var locationCallback: LocationCallback

    override fun onCreate() {
        super.onCreate()
        fusedLocationClient = LocationServices.getFusedLocationProviderClient(this)

        locationCallback = object : LocationCallback() {
            override fun onLocationResult(locationResult: LocationResult) {
                locationResult.lastLocation?.let { location ->
                    updateCountryFromLocation(location)
                }
            }
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        startForegroundNotification()
        requestLocationUpdates()
        return START_STICKY
    }

    private fun startForegroundNotification() {
        val notification = NotificationCompat.Builder(this, WorkshopTechApp.CHANNEL_SYSTEM)
            .setSmallIcon(android.R.drawable.ic_menu_compass)
            .setContentTitle("تحديد الموقع")
            .setContentText("جاري تحديد الدولة تلقائياً...")
            .setPriority(NotificationCompat.PRIORITY_LOW)
            .build()

        startForeground(1001, notification)
    }

    private fun requestLocationUpdates() {
        if (ContextCompat.checkSelfPermission(
                this, Manifest.permission.ACCESS_FINE_LOCATION
            ) == PackageManager.PERMISSION_GRANTED
        ) {
            val locationRequest = LocationRequest.Builder(
                Priority.PRIORITY_BALANCED_POWER_ACCURACY, 3600000
            ).setMinUpdateIntervalMillis(1800000).build()

            fusedLocationClient.requestLocationUpdates(
                locationRequest, locationCallback, Looper.getMainLooper()
            )
        }
    }

    private fun updateCountryFromLocation(location: Location) {
        val country = when {
            location.latitude in 19.0..34.0 && location.longitude in 9.0..26.0 -> "LY"

            location.latitude in 24.5..26.5 && location.longitude in 50.5..51.5 -> "QA"
            location.latitude in 25.5..27.0 && location.longitude in 50.0..51.0 -> "BH"
            location.latitude in 16.0..27.0 && location.longitude in 52.0..60.0 -> "OM"
            location.latitude in 29.0..33.5 && location.longitude in 34.0..39.5 -> "JO"
            location.latitude in 33.0..35.0 && location.longitude in 35.0..36.5 -> "LB"
            location.latitude in 32.0..37.5 && location.longitude in 35.5..42.5 -> "SY"
            location.latitude in 29.0..37.5 && location.longitude in 38.5..48.5 -> "IQ"

        DamageFindingEntity::class,
        InspectionEntity::class,
        WorkflowTaskEntity::class,
        TechnicianEntity::class,
        InventoryEntity::class,
        AnalysisResultEntity::class
    ],
    version = 3,
    exportSchema = false
)
abstract class AppDatabase : RoomDatabase() {
    abstract fun caseDao(): CaseDao
    abstract fun customerDao(): CustomerDao
    abstract fun photoDao(): PhotoDao
    abstract fun inspectionDao(): InspectionDao
    abstract fun workflowTaskDao(): WorkflowTaskDao
    abstract fun technicianDao(): TechnicianDao
    abstract fun inventoryDao(): InventoryDao
    abstract fun analysisResultDao(): AnalysisResultDao
    abstract fun damageFindingDao(): DamageFindingDao

    companion object {
        @Volatile
        private var INSTANCE: AppDatabase? = null

        fun getInstance(context: Context): AppDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    "workshop_tech.db"
                )
                .fallbackToDestructiveMigration()
                .addCallback(DatabaseCallback())
                .build()
                INSTANCE = instance
                instance
            }
        }
    }

    class DatabaseCallback : Callback() {
        override fun onCreate(db: androidx.sqlite.db.SupportSQLiteDatabase) {
            super.onCreate(db)
            db.execSQL("CREATE INDEX IF NOT EXISTS idx_cases_status_created ON cases(status, createdAt)")
            db.execSQL("CREATE INDEX IF NOT EXISTS idx_photos_case_captured ON case_photos(caseId, capturedAt)")
            db.execSQL("CREATE INDEX IF NOT EXISTS idx_inspections_case_type ON inspections(caseId, type)")
            db.execSQL("CREATE INDEX IF NOT EXISTS idx_analysis_photo_layer ON analysis_results(photoId, layer)")
        }
    }
}
