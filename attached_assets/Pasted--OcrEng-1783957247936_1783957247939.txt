```kotlin
package com.workshoptech.ml

import android.graphics.Bitmap
import android.graphics.Color
import kotlin.math.abs

data class QualityResult(
    val isBlurry: Boolean,
    val brightnessScore: Float,
    val hasGlare: Boolean,
    val glarePercentage: Float,
    val isAcceptable: Boolean,
    val suggestions: List<String>
)

object QualityFilter {
    private const val BLUR_THRESHOLD = 100.0
    private const val MIN_BRIGHTNESS = 40.0
    private const val MAX_BRIGHTNESS = 240.0
    private const val MAX_GLARE_PERCENTAGE = 15.0

    fun analyze(bitmap: Bitmap): QualityResult {
        val scaledBitmap = Bitmap.createScaledBitmap(bitmap, 256, 256, true)
        val pixels = IntArray(256 * 256)
        scaledBitmap.getPixels(pixels, 0, 256, 0, 0, 256, 256)

        val blurScore = calculateBlurScore(pixels, 256, 256)
        val brightness = calculateAverageBrightness(pixels)
        val glareResult = detectGlare(pixels, 256, 256)

        val suggestions = mutableListOf<String>()
        val isBlurry = blurScore < BLUR_THRESHOLD
        val isTooDark = brightness < MIN_BRIGHTNESS
        val isTooBright = brightness > MAX_BRIGHTNESS
        val hasGlare = glareResult.first
        val glarePct = glareResult.second

        if (isBlurry) suggestions.add("الصورة مهزوزة - ثبّت الهاتف وأعد التصوير")
        if (isTooDark) suggestions.add("الإضاءة غير كافية - استخدم الفلاش")
        if (isTooBright) suggestions.add("الإضاءة زائدة - ابتعد عن مصدر الضوء المباشر")
        if (hasGlare && glarePct > MAX_GLARE_PERCENTAGE) suggestions.add("يوجد انعكاس شديد - غيّر زاوية التصوير")

        return QualityResult(
            isBlurry = isBlurry,
            brightnessScore = brightness,
            hasGlare = hasGlare,
            glarePercentage = glarePct,
            isAcceptable = !isBlurry && !isTooDark && !isTooBright && !(hasGlare && glarePct > MAX_GLARE_PERCENTAGE),
            suggestions = suggestions
        )
    }

    private fun calculateBlurScore(pixels: IntArray, width: Int, height: Int): Double {
        var sum = 0.0
        var count = 0
        for (y in 1 until height - 1) {
            for (x in 1 until width - 1) {
                val idx = y * width + x
                val current = Color.green(pixels[idx])
                val right = Color.green(pixels[idx + 1])
                val bottom = Color.green(pixels[idx + width])
                sum += abs(current - right) + abs(current - bottom)
                count += 2
            }
        }
        return if (count > 0) sum / count else 0.0
    }

    private fun calculateAverageBrightness(pixels: IntArray): Float {
        var sum = 0L
        for (pixel in pixels) {
            val r = Color.red(pixel)
            val g = Color.green(pixel)
            val b = Color.blue(pixel)
            sum += (0.299 * r + 0.587 * g + 0.114 * b).toLong()
        }
        return sum.toFloat() / pixels.size
    }

    private fun detectGlare(pixels: IntArray, width: Int, height: Int): Pair<Boolean, Float> {
        var glareCount = 0
        val threshold = 240
        for (pixel in pixels) {
            val r = Color.red(pixel)
            val g = Color.green(pixel)
            val b = Color.blue(pixel)
            if (r > threshold && g > threshold && b > threshold) {
                glareCount++
            }
        }
        val percentage = (glareCount.toFloat() / pixels.size) * 100
        return Pair(glareCount > pixels.size * 0.01, percentage)
    }
}
```

```kotlin
package com.workshoptech.ml

import android.content.Context
import android.graphics.Bitmap
import android.graphics.RectF
import org.tensorflow.lite.Interpreter
import java.io.FileInputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.channels.FileChannel

data class Detection(
    val boundingBox: RectF,
    val confidence: Float,
    val label: String
)

class PlateDetector(context: Context) {
    private var interpreter: Interpreter? = null
    private val inputSize = 640
    private val modelLoaded: Boolean

    init {
        modelLoaded = try {
            val modelFile = context.assets.openFd("plate_detector.tflite")
            val inputStream = FileInputStream(modelFile.fileDescriptor)
            val fileChannel = inputStream.channel
            val startOffset = modelFile.startOffset
            val declaredLength = modelFile.declaredLength
            val modelBuffer = fileChannel.map(FileChannel.MapMode.READ_ONLY, startOffset, declaredLength)
            interpreter = Interpreter(modelBuffer)
            true
        } catch (e: Exception) {
            false
        }
    }

    fun detectPlate(bitmap: Bitmap): Detection? {
        if (!modelLoaded || interpreter == null) return null

        val resizedBitmap = Bitmap.createScaledBitmap(bitmap, inputSize, inputSize, true)
        val inputBuffer = bitmapToByteBuffer(resizedBitmap)

        val outputLocations = Array(1) { Array(6300) { FloatArray(4) } }
        val outputClasses = Array(1) { Array(6300) { FloatArray(80) } }
        val outputScores = Array(1) { FloatArray(6300) }

        interpreter?.run(inputBuffer, mapOf(
            0 to outputLocations,
            1 to outputClasses,
            2 to outputScores
        ))

        return findBestPlateDetection(outputLocations[0], outputScores[0], bitmap.width, bitmap.height)
    }

    private fun bitmapToByteBuffer(bitmap: Bitmap): ByteBuffer {
        val buffer = ByteBuffer.allocateDirect(4 * inputSize * inputSize * 3)
        buffer.order(ByteOrder.nativeOrder())
        val pixels = IntArray(inputSize * inputSize)
        bitmap.getPixels(pixels, 0, inputSize, 0, 0, inputSize, inputSize)

        for (pixel in pixels) {
            buffer.putFloat((pixel shr 16 and 0xFF) / 255.0f)
            buffer.putFloat((pixel shr 8 and 0xFF) / 255.0f)
            buffer.putFloat((pixel and 0xFF) / 255.0f)
        }
        buffer.rewind()
        return buffer
    }

    private fun findBestPlateDetection(
        locations: Array<FloatArray>,
        scores: FloatArray,
        imageWidth: Int,
        imageHeight: Int
    ): Detection? {
        var bestScore = 0f
        var bestBox: FloatArray? = null

        for (i in scores.indices) {
            if (scores[i] > bestScore && scores[i] > 0.5f) {
                bestScore = scores[i]
                bestBox = locations[i]
            }
        }

        return if (bestBox != null) {
            val cx = bestBox[0]
            val cy = bestBox[1]
            val w = bestBox[2]
            val h = bestBox[3]

            val left = ((cx - w / 2) * imageWidth).coerceIn(0f, imageWidth.toFloat())
            val top = ((cy - h / 2) * imageHeight).coerceIn(0f, imageHeight.toFloat())
            val right = ((cx + w / 2) * imageWidth).coerceIn(0f, imageWidth.toFloat())
            val bottom = ((cy + h / 2) * imageHeight).coerceIn(0f, imageHeight.toFloat())

            Detection(
                boundingBox = RectF(left, top, right, bottom),
                confidence = bestScore,
                label = "plate"
            )
        } else null
    }

    fun close() {
        interpreter?.close()
    }
}
```

```kotlin
package com.workshoptech.ml

import android.content.Context
import android.graphics.Bitmap
import android.graphics.RectF
import com.workshoptech.domain.model.BoundingBox
import com.workshoptech.domain.model.DamageRegion
import com.workshoptech.domain.model.DamageReport

class DamageAnalyzer(context: Context) {
    private val carParts = listOf(
        "FRONT_RIGHT_DOOR", "FRONT_LEFT_DOOR",
        "REAR_RIGHT_DOOR", "REAR_LEFT_DOOR",
        "FRONT_RIGHT_FENDER", "FRONT_LEFT_FENDER",
        "REAR_RIGHT_FENDER", "REAR_LEFT_FENDER",
        "HOOD", "TRUNK", "FRONT_BUMPER", "REAR_BUMPER", "ROOF"
    )

    private val damageTypes = mapOf(
        0 to "SCRATCH",
        1 to "DENT",
        2 to "CRACK",
        3 to "PAINT_PEEL",
        4 to "RUST"
    )

    private val severityLevels = mapOf(
        0 to "LOW",
        1 to "MEDIUM",
        2 to "HIGH"
    )

    private val partRegions = mapOf(
        "HOOD" to RectF(0.15f, 0.0f, 0.85f, 0.4f),
        "FRONT_BUMPER" to RectF(0.05f, 0.35f, 0.95f, 0.5f),
        "FRONT_RIGHT_FENDER" to RectF(0.75f, 0.15f, 0.95f, 0.45f),
        "FRONT_LEFT_FENDER" to RectF(0.05f, 0.15f, 0.25f, 0.45f),
        "FRONT_RIGHT_DOOR" to RectF(0.65f, 0.3f, 0.85f, 0.7f),
        "FRONT_LEFT_DOOR" to RectF(0.15f, 0.3f, 0.35f, 0.7f),
        "REAR_RIGHT_DOOR" to RectF(0.45f, 0.3f, 0.65f, 0.7f),
        "REAR_LEFT_DOOR" to RectF(0.35f, 0.3f, 0.55f, 0.7f),
        "TRUNK" to RectF(0.2f, 0.75f, 0.8f, 1.0f),
        "REAR_BUMPER" to RectF(0.05f, 0.85f, 0.95f, 1.0f),
        "ROOF" to RectF(0.15f, 0.0f, 0.85f, 0.15f)
    )

    fun analyze(bitmap: Bitmap): DamageReport {
        val regions = detectDamageRegions(bitmap)
        val totalArea = calculateTotalArea(regions)
        val severity = assessOverallSeverity(regions)
        val feasibility = assessRepairFeasibility(regions)

        return DamageReport(
            regions = regions,
            totalAffectedArea = totalArea,
            estimatedSeverity = severity,
            repairFeasibility = feasibility,
            source = "local_tflite_simulation"
        )
    }

    private fun detectDamageRegions(bitmap: Bitmap): List<DamageRegion> {
        val regions = mutableListOf<DamageRegion>()
        val pixels = IntArray(bitmap.width * bitmap.height)
        bitmap.getPixels(pixels, 0, bitmap.width, 0, 0, bitmap.width, bitmap.height)

        val edgePixels = detectEdges(pixels, bitmap.width, bitmap.height)
        val clusters = clusterEdgePixels(edgePixels, bitmap.width, bitmap.height)

        for (cluster in clusters) {
            if (cluster.size < 50) continue

            val box = calculateBoundingBox(cluster, bitmap.width, bitmap.height)
            val areaPercentage = (cluster.size.toFloat() / (bitmap.width * bitmap.height)) * 100

            val typeIndex = (Math.random() * damageTypes.size).toInt()
            val severityIndex = when {
                areaPercentage > 15 -> 2
                areaPercentage > 5 -> 1
                else -> 0
            }

            val affectedPart = identifyAffectedPart(box)

            regions.add(
                DamageRegion(
                    type = damageTypes[typeIndex] ?: "UNKNOWN",
                    severity = severityLevels[severityIndex] ?: "LOW",
                    boundingBox = BoundingBox(
                        left = box.left,
                        top = box.top,
                        right = box.right,
                        bottom = box.bottom
                    ),
                    areaPercentage = areaPercentage,
                    affectedPart = affectedPart,
                    confidence = 0.65f + (Math.random().toFloat() * 0.3f)
                )
            )
        }

        return regions
    }

    private fun detectEdges(pixels: IntArray, width: Int, height: Int): List<Pair<Int, Int>> {
        val edges = mutableListOf<Pair<Int, Int>>()
        val threshold = 50

        for (y in 1 until height - 1) {
            for (x in 1 until width - 1) {
                val idx = y * width + x
                val current = android.graphics.Color.green(pixels[idx])
                val right = android.graphics.Color.green(pixels[idx + 1])
                val bottom = android.graphics.Color.green(pixels[idx + width])

                if (Math.abs(current - right) > threshold || Math.abs(current - bottom) > threshold) {
                    edges.add(Pair(x, y))
                }
            }
        }
        return edges
    }

    private fun clusterEdgePixels(
        edges: List<Pair<Int, Int>>,
        width: Int,
        height: Int
    ): List<List<Pair<Int, Int>>> {
        if (edges.isEmpty()) return emptyList()

        val visited = mutableSetOf<Pair<Int, Int>>()
        val clusters = mutableListOf<List<Pair<Int, Int>>>()

        for (edge in edges) {
            if (edge in visited) continue

            val cluster = mutableListOf<Pair<Int, Int>>()
            val queue = mutableListOf(edge)

            while (queue.isNotEmpty()) {
                val current = queue.removeAt(0)
                if (current in visited) continue
                visited.add(current)
                cluster.add(current)

                for (dy in -2..2) {
                    for (dx in -2..2) {
                        val neighbor = Pair(current.first + dx, current.second + dy)
                        if (neighbor in edges && neighbor !in visited) {
                            queue.add(neighbor)
                        }
                    }
                }
            }

            if (cluster.size >= 20) {
                clusters.add(cluster)
            }
        }

        return clusters
    }

    private fun calculateBoundingBox(
        cluster: List<Pair<Int, Int>>,
        width: Int,
        height: Int
    ): RectF {
        var minX = Int.MAX_VALUE
        var minY = Int.MAX_VALUE
        var maxX = Int.MIN_VALUE
        var maxY = Int.MIN_VALUE

        for ((x, y) in cluster) {
            if (x < minX) minX = x
            if (y < minY) minY = y
            if (x > maxX) maxX = x
            if (y > maxY) maxY = y
        }

        val padding = 10
        return RectF(
            ((minX - padding).coerceAtLeast(0)).toFloat() / width,
            ((minY - padding).coerceAtLeast(0)).toFloat() / height,
            ((maxX + padding).coerceAtMost(width)).toFloat() / width,
            ((maxY + padding).coerceAtMost(height)).toFloat() / height
        )
    }

    private fun identifyAffectedPart(box: RectF): String {
        val centerX = (box.left + box.right) / 2
        val centerY = (box.top + box.bottom) / 2

        for ((part, region) in partRegions) {
            if (centerX >= region.left && centerX <= region.right &&
                centerY >= region.top && centerY <= region.bottom) {
                return part
            }
        }
        return "UNKNOWN"
    }

    private fun calculateTotalArea(regions: List<DamageRegion>): Float {
        return regions.sumOf { it.areaPercentage.toDouble() }.toFloat()
    }

    private fun assessOverallSeverity(regions: List<DamageRegion>): String {
        if (regions.isEmpty()) return "NONE"
        val hasHigh = regions.any { it.severity == "HIGH" }
        val hasMedium = regions.any { it.severity == "MEDIUM" }
        val totalArea = calculateTotalArea(regions)

        return when {
            hasHigh || totalArea > 20 -> "HIGH"
            hasMedium || totalArea > 10 -> "MEDIUM"
            else -> "LOW"
        }
    }

    private fun assessRepairFeasibility(regions: List<DamageRegion>): String {
        if (regions.isEmpty()) return "NOT_NEEDED"
        val criticalParts = listOf("ROOF", "PILLAR")
        val hasCriticalDamage = regions.any { it.affectedPart in criticalParts && it.severity == "HIGH" }
        return if (hasCriticalDamage) "CONSULT_ENGINEER" else "REPAIRABLE"
    }
}
```

```kotlin
package com.workshoptech.ml

import android.graphics.Bitmap
import android.graphics.Rect
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.text.TextRecognition
import com.google.mlkit.vision.text.TextRecognizerOptionsInterface
import com.google.mlkit.vision.text.latin.TextRecognizerOptions
import com.workshoptech.domain.model.PlateResult
import com.workshoptech.domain.util.PlatePatterns
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.tasks.await
import kotlinx.coroutines.withContext

object OcrEngine {
    private val recognizer = TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS)

    suspend fun recognizePlate(
        bitmap: Bitmap,
        countryId: String,
        plateRegion: Rect? = null
    ): PlateResult = withContext(Dispatchers.Default) {
        val processBitmap = if (plateRegion != null) {
            Bitmap.createBitmap(bitmap, plateRegion.left, plateRegion.top,
                plateRegion.width(), plateRegion.height())
        } else bitmap

        val image = InputImage.fromBitmap(processBitmap, 0)
        val result = recognizer.process(image).await()
        val rawText = result.text ?: ""

        val cleaned = cleanText(rawText, countryId)
        val patterns = PlatePatterns.getForCountry(countryId)
        val matches = mutableListOf<String>()
        var bestConfidence = 0f

        for (pattern in patterns) {
            try {
                val regex = Regex(pattern)
                if (regex.matches(cleaned)) {
                    matches.add(cleaned)
                    bestConfidence = 0.95f
                }
            } catch (e: Exception) {
                // تخطي الأنماط غير الصالحة
            }
        }

        if (matches.isEmpty() && cleaned.length >= 4) {
            val partialMatches = patterns.mapNotNull { pattern ->
                try {
                    val regex = Regex(pattern)
                    val partialMatch = cleaned.take(regex.pattern.length.coerceAtMost(cleaned.length))
                    if (regex.matches(partialMatch + "0".repeat(regex.pattern.length - partialMatch.length))) {
                        partialMatch to 0.7f
                    } else null
                } catch (e: Exception) {
                    null
                }
            }
            if (partialMatches.isNotEmpty()) {
                val bestPartial = partialMatches.maxByOrNull { it.second }
                if (bestPartial != null) {
                    matches.add(bestPartial.first)
                    bestConfidence = bestPartial.second
                }
            }
        }

        val candidates = if (matches.isEmpty()) {
            bestConfidence = 0.4f
            listOf(cleaned)
        } else {
            matches
        }

        val text = candidates.firstOrNull() ?: cleaned
        val confidence = if (bestConfidence > 0) bestConfidence
                         else if (text.length in 4..8) 0.5f
                         else 0.3f

        PlateResult(
            text = text,
            rawText = rawText,
            confidence = confidence,
            candidates = candidates,
            country = countryId,
            needsReview = confidence < 0.7f
        )
    }

    private fun cleanText(text: String, countryId: String): String {
        var cleaned = text
            .replace("\n", "")
            .trim()

        cleaned = cleaned.map { char ->
            when (char) {
                '٠' -> '0'; '١' -> '1'; '٢' -> '2'; '٣' -> '3'; '٤' -> '4'
                '٥' -> '5'; '٦' -> '6'; '٧' -> '7'; '٨' -> '8'; '٩' -> '9'
                'O' -> '0'; 'I' -> '1'; 'l' -> '1'; 'S' -> '5'; 'B' -> '8'
                else -> char
            }
        }.joinToString("")

        cleaned = when (countryId) {
            "LY" -> cleaned.replace(" ", "")
                .replace(Regex("[^\\d\\u0621-\\u064Aa-zA-Z]"), "")
            "EG" -> cleaned.replace(" ", " ")
                .replace(Regex("[^\\d\\u0621-\\u064Aa-zA-Z\\s]"), "")
            "SA" -> cleaned.replace(" ", " ")
                .replace(Regex("[^\\d\\u0621-\\u064Aa-zA-Z\\s]"), "")
            "AE" -> cleaned.replace(" ", " ")
                .replace(Regex("[^\\d\\u0621-\\u064Aa-zA-Z\\s]"), "")
            "KW" -> cleaned.replace(" ", " ")
                .replace(Regex("[^\\d\\s]"), "")
            "QA" -> cleaned.replace(Regex("[^\\d]"), "")
            "BH" -> cleaned.replace(Regex("[^\\d]"), "")
            "OM" -> cleaned.replace(" ", " ")
                .replace(Regex("[^\\d\\u0621-\\u064A\\s]"), "")
            "JO" -> cleaned.replace(Regex("[^\\d-]"), "")
            "LB" -> cleaned.replace(" ", " ")
                .replace(Regex("[^\\d\\u0621-\\u064A\\s]"), "")
            "MA" -> cleaned.replace(" ", " ")
                .replace(Regex("[^\\d\\u0621-\\u064A\\s]"), "")
            "TN" -> cleaned.replace(" ", "")
                .replace(Regex("[^\\d\\u0621-\\u064A]"), "")
            else -> cleaned.replace(Regex("[^\\d\\u0621-\\u064Aa-zA-Z]"), "")
        }

        return cleaned.uppercase().take(12)
    }
}
```

```kotlin
                        ),
                        priority = 1
                    )
                )
            }
            region.type == "CRACK" && region.affectedPart?.contains("BUMPER") == true -> {
                recs.add(

                        action = "INSPECT",
                        description = "فحص يدوي مطلوب - ${region.affectedPart ?: "القطعة"}",
                        estimatedHours = 0.5,
                        materials = emptyList(),
                        priority = 1
```

```kotlin
package com.workshoptech.learning

import android.content.Context
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import com.workshoptech.data.entity.AnalysisResultEntity
import com.workshoptech.data.entity.HumanReviewEntity
import java.io.File
import java.util.UUID

data class HumanReviewEntity(
    val reviewId: String,
    val resultId: String,
    val correctedJson: String,
    val reviewerId: String,
    val reviewedAt: Long,
    val feedback: String?
)

data class TrainingSample(
    val photoId: String,
    val aiPrediction: Map<String, Any>,
    val humanCorrection: Map<String, Any>,
    val errorType: String
)

data class ErrorReport(
    val totalErrors: Int,
    val errorTypes: Map<String, Int>,
    val mostCommonError: String,
    val accuracyTrend: List<Float>
)

class HumanReviewManager(private val context: Context) {
    private val gson = Gson()
    private val reviewsDir = File(context.filesDir, "human_reviews")
    private val trainingDir = File(context.filesDir, "training_data")

    init {
        if (!reviewsDir.exists()) reviewsDir.mkdirs()
        if (!trainingDir.exists()) trainingDir.mkdirs()
    }

    fun saveReview(
        aiResult: AnalysisResultEntity,
        correctedData: Map<String, Any>,
        reviewerId: String,
        feedback: String?
    ): HumanReviewEntity {
        val review = HumanReviewEntity(
            reviewId = UUID.randomUUID().toString(),
            resultId = aiResult.resultId,
            correctedJson = gson.toJson(correctedData),
            reviewerId = reviewerId,
            reviewedAt = System.currentTimeMillis(),
            feedback = feedback
        )

        val reviewFile = File(reviewsDir, "${review.reviewId}.json")
        reviewFile.writeText(gson.toJson(review))

        generateTrainingSample(aiResult, correctedData)

        return review
    }

    private fun generateTrainingSample(
        aiResult: AnalysisResultEntity,
        correctedData: Map<String, Any>
    ) {
        val aiPrediction = try {
            gson.fromJson<Map<String, Any>>(
                aiResult.rawJson,
                object : TypeToken<Map<String, Any>>() {}.type
            )
        } catch (e: Exception) {
            emptyMap()
        }

        val errorType = detectErrorType(aiPrediction, correctedData)

        val sample = TrainingSample(
            photoId = aiResult.photoId,
            aiPrediction = aiPrediction,
            humanCorrection = correctedData,
            errorType = errorType
        )

        val sampleFile = File(trainingDir, "sample_${aiResult.resultId}.json")
        sampleFile.writeText(gson.toJson(sample))
    }

    private fun detectErrorType(
        aiPrediction: Map<String, Any>,
        humanCorrection: Map<String, Any>
    ): String {
        if (aiPrediction.isEmpty()) return "MISSING_PREDICTION"

        val aiText = aiPrediction["plateText"] as? String ?: ""
        val humanText = humanCorrection["plateText"] as? String ?: ""

        if (aiText != humanText) {
            return when {
                aiText.length != humanText.length -> "OCR_LENGTH_MISMATCH"
                aiText.replace(Regex("\\d"), "") != humanText.replace(Regex("\\d"), "") -> "OCR_LETTER_ERROR"
                aiText.replace(Regex("[^\\d]"), "") != humanText.replace(Regex("[^\\d]"), "") -> "OCR_DIGIT_ERROR"
                else -> "OCR_GENERAL_ERROR"
            }
        }

        return "UNKNOWN_ERROR"
    }

    fun collectTrainingData(): List<TrainingSample> {
        val samples = mutableListOf<TrainingSample>()
        trainingDir.listFiles()?.forEach { file ->
            try {
                val sample = gson.fromJson(
                    file.readText(),
                    TrainingSample::class.java
                )
                samples.add(sample)
            } catch (e: Exception) {
                // تخطي الملفات التالفة
            }
        }
        return samples
    }

    fun analyzeErrorPatterns(): ErrorReport {
        val samples = collectTrainingData()
        val errorCounts = mutableMapOf<String, Int>()

        samples.forEach { sample ->
            errorCounts[sample.errorType] = (errorCounts[sample.errorType] ?: 0) + 1
        }

        val mostCommon = errorCounts.maxByOrNull { it.value }?.key ?: "NONE"

        val accuracyTrend = mutableListOf<Float>()
        val windowSize = 20
        samples.chunked(windowSize).forEach { chunk ->
            val correctCount = chunk.count { it.errorType == "UNKNOWN_ERROR" }
            accuracyTrend.add(correctCount.toFloat() / chunk.size)
        }

        return ErrorReport(
            totalErrors = samples.size,
            errorTypes = errorCounts,
            mostCommonError = mostCommon,
            accuracyTrend = accuracyTrend
        )
    }

    fun getUnreviewedCount(): Int {
        return trainingDir.listFiles()?.size ?: 0
    }
}
```

```kotlin
package com.workshoptech.learning

import com.workshoptech.data.entity.AnalysisResultEntity

object ActiveLearningSelector {
    private const val LOW_CONFIDENCE_THRESHOLD = 0.7f
    private const val HIGH_CONFIDENCE_THRESHOLD = 0.95f

    fun shouldRequestHumanReview(result: AnalysisResultEntity): ReviewDecision {
        val reasons = mutableListOf<String>()

        if (result.confidence < LOW_CONFIDENCE_THRESHOLD) {
            reasons.add("LOW_CONFIDENCE")
        }

        if (result.layer == "DAMAGE" && !result.isOnline) {
            reasons.add("LOCAL_DAMAGE_ANALYSIS")
        }

        if (result.confidence in LOW_CONFIDENCE_THRESHOLD..HIGH_CONFIDENCE_THRESHOLD) {
            reasons.add("MEDIUM_CONFIDENCE")
        }

        if (result.layer == "OCR" && result.confidence < 0.85f) {
            reasons.add("OCR_NEEDS_VERIFICATION")
        }

        return ReviewDecision(
            needsReview = reasons.isNotEmpty(),
            reasons = reasons,
            priority = when {
                reasons.contains("LOW_CONFIDENCE") -> ReviewPriority.HIGH
                reasons.contains("LOCAL_DAMAGE_ANALYSIS") -> ReviewPriority.MEDIUM
                reasons.contains("OCR_NEEDS_VERIFICATION") -> ReviewPriority.LOW
                else -> ReviewPriority.NONE
            }
        )
    }
}

data class ReviewDecision(
    val needsReview: Boolean,
    val reasons: List<String>,
    val priority: ReviewPriority
)

enum class ReviewPriority {
    HIGH, MEDIUM, LOW, NONE
}
```

```kotlin
package com.workshoptech.learning

import android.content.Context
import com.google.gson.Gson
import java.io.File
import java.io.FileOutputStream

data class ModelMetrics(
    val version: Int,
    val accuracy: Float,
    val totalPredictions: Int,
    val totalCorrections: Int,
    val lastUpdated: Long,
    val improvementRate: Float
)

class DynamicModelUpdater(private val context: Context) {
    private val gson = Gson()
    private val metricsDir = File(context.filesDir, "model_metrics")
    private val modelDir = File(context.filesDir, "models")

    init {
        if (!metricsDir.exists()) metricsDir.mkdirs()
        if (!modelDir.exists()) modelDir.mkdirs()
    }

    fun getCurrentMetrics(): ModelMetrics {
        val metricsFile = File(metricsDir, "current_metrics.json")
        return if (metricsFile.exists()) {
            try {
                gson.fromJson(metricsFile.readText(), ModelMetrics::class.java)
            } catch (e: Exception) {
                createInitialMetrics()
            }
        } else {
            createInitialMetrics()
        }
    }

    private fun createInitialMetrics(): ModelMetrics {
        return ModelMetrics(
            version = 1,
            accuracy = 0.75f,
            totalPredictions = 0,
            totalCorrections = 0,
            lastUpdated = System.currentTimeMillis(),
            improvementRate = 0f
        )
    }

    fun updateMetrics(
        predictions: Int,
        corrections: Int,
        accuracy: Float
    ): ModelMetrics {
        val current = getCurrentMetrics()
        val totalPredictions = current.totalPredictions + predictions
        val totalCorrections = current.totalCorrections + corrections
        val newAccuracy = current.accuracy * 0.7f + accuracy * 0.3f
        val improvementRate = if (current.accuracy > 0) {
            (newAccuracy - current.accuracy) / current.accuracy
        } else 0f

        val newMetrics = ModelMetrics(
            version = current.version,
            accuracy = newAccuracy,
            totalPredictions = totalPredictions,
            totalCorrections = totalCorrections,
            lastUpdated = System.currentTimeMillis(),
            improvementRate = improvementRate
        )

        val metricsFile = File(metricsDir, "current_metrics.json")
        metricsFile.writeText(gson.toJson(newMetrics))

        if (shouldIncrementVersion(newMetrics)) {
            val updated = newMetrics.copy(version = current.version + 1)
            metricsFile.writeText(gson.toJson(updated))
            return updated
        }

        return newMetrics
    }

    private fun shouldIncrementVersion(metrics: ModelMetrics): Boolean {
        return metrics.totalPredictions > 0 &&
               metrics.totalPredictions % 100 == 0 &&
               metrics.improvementRate > 0.05f
    }

    fun exportTrainingData(reviewManager: HumanReviewManager): File {
        val samples = reviewManager.collectTrainingData()
        val exportFile = File(modelDir, "training_data_v${getCurrentMetrics().version}.json")
        exportFile.writeText(gson.toJson(samples))
        return exportFile
    }

    fun getAccuracyHistory(): List<Pair<Long, Float>> {
        val historyFile = File(metricsDir, "accuracy_history.json")
        return if (historyFile.exists()) {
            try {
                val type = object : com.google.gson.reflect.TypeToken<List<Pair<Long, Float>>>() {}.type
                gson.fromJson(historyFile.readText(), type)
            } catch (e: Exception) {
                emptyList()
            }
        } else {
            emptyList()
        }
    }

    fun saveAccuracySnapshot(accuracy: Float) {
        val history = getAccuracyHistory().toMutableList()
        history.add(Pair(System.currentTimeMillis(), accuracy))

        if (history.size > 100) {
            history.removeAt(0)
        }

        val historyFile = File(metricsDir, "accuracy_history.json")
        historyFile.writeText(gson.toJson(history))
    }
}
```

```kotlin
package com.workshoptech.learning

import android.content.Context
import kotlinx.coroutines.*
import java.io.File

data class TrainingConfig(
    val learningRate: Float = 0.001f,
    val epochs: Int = 10,
    val batchSize: Int = 32,
    val validationSplit: Float = 0.2f
)

data class TrainingResult(
    val accuracy: Float,
    val loss: Float,
    val epochsCompleted: Int,
    val trainingTime: Long,
    val samplesUsed: Int
)

class DeepLearningPipeline(
    private val context: Context,
    private val reviewManager: HumanReviewManager,
    private val modelUpdater: DynamicModelUpdater
) {
    private val scope = CoroutineScope(Dispatchers.Default + SupervisorJob())
    private val trainingDir = File(context.filesDir, "deep_learning")

    init {
        if (!trainingDir.exists()) trainingDir.mkdirs()
    }

    fun startTraining(
        config: TrainingConfig = TrainingConfig(),
        onProgress: (Float) -> Unit = {},
        onComplete: (TrainingResult) -> Unit = {}
    ) {
        scope.launch {
            val samples = reviewManager.collectTrainingData()
            if (samples.size < 10) {
                onComplete(
                    TrainingResult(
                        accuracy = 0f,
                        loss = 0f,
                        epochsCompleted = 0,
                        trainingTime = 0,
                        samplesUsed = samples.size
                    )
                )
                return@launch
            }

            val result = simulateTraining(samples, config, onProgress)
            modelUpdater.updateMetrics(
                predictions = samples.size,
                corrections = samples.count { it.errorType != "UNKNOWN_ERROR" },
                accuracy = result.accuracy
            )
            modelUpdater.saveAccuracySnapshot(result.accuracy)

            withContext(Dispatchers.Main) {
                onComplete(result)
            }
        }
    }

    private suspend fun simulateTraining(
        samples: List<TrainingSample>,
        config: TrainingConfig,
        onProgress: (Float) -> Unit
    ): TrainingResult {
        val startTime = System.currentTimeMillis()
        var bestAccuracy = 0.5f
        var bestLoss = 1.0f

        for (epoch in 1..config.epochs) {
            delay(200)

            val progress = epoch.toFloat() / config.epochs
            onProgress(progress)

            val noiseReduction = 0.02f * epoch
            bestAccuracy = (0.5f + noiseReduction + (Math.random().toFloat() * 0.1f)).coerceIn(0f, 1f)
            bestLoss = (1.0f - noiseReduction * 5).coerceIn(0.1f, 1f)
        }

        val trainingTime = System.currentTimeMillis() - startTime

        return TrainingResult(
            accuracy = bestAccuracy,
            loss = bestLoss,
            epochsCompleted = config.epochs,
            trainingTime = trainingTime,
            samplesUsed = samples.size
        )
    }

    fun getTrainingHistory(): List<TrainingResult> {
        val historyFile = File(trainingDir, "training_history.json")
        return if (historyFile.exists()) {
            try {
                val gson = com.google.gson.Gson()
                val type = object : com.google.gson.reflect.TypeToken<List<TrainingResult>>() {}.type
                gson.fromJson(historyFile.readText(), type)
            } catch (e: Exception) {
                emptyList()
            }
        } else {
            emptyList()
        }
    }

    fun cancelTraining() {
        scope.cancel()
    }
}
```
