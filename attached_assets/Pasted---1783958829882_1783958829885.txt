// ml/PlateOCR.kt
data class PlateResult(
    val text: String,
    val confidence: Float,
    val candidates: List<String>,  // بدائل محتملة
    val country: String?,          // LY/EG تم تخمينها
    val needsReview: Boolean       // true إذا الثقة منخفضة
)

class PlateOCR {
    suspend fun readPlate(plateBitmap: Bitmap): PlateResult {
        // 1. ML Kit OCR
        val rawText = runOcr(plateBitmap)
        
        // 2. تنظيف وتطبيع
        val cleaned = PlateNormalizer.clean(rawText)
        
        // 3. قواعد ليبيا/Mصر
        val (country, candidates) = PlateMatcher.match(cleaned)
        
        // 4. تحديد إذا كانت تحتاج مراجعة بشرية
        val needsReview = candidates.size > 1 || confidence < 0.8
        
        return PlateResult(cleaned, confidence, candidates, country, needsReview)
    }
}

المحور السادس (اختياري متقدم): وضع الأونلاين الهجين

عند توفر الإنترنت، يضاف تحسين صامت:

```kotlin
// workers/OnlineEnhanceWorker.kt
class OnlineEnhanceWorker(
    context: Context,
    params: WorkerParameters
) : CoroutineWorker(context, params) {
    
    override suspend fun doWork(): Result {
        val photoId = inputData.getString("photoId") ?: return Result.failure()
        
        // 1. تحميل الصورة
        val photo = db.photoDao().getById(photoId)
        
        // 2. إرسال القصاصات فقط (للخصوصية والتكلفة)
        val plateCrop = cropPlate(photo.imagePath)
        val logoCrop = cropLogo(photo.imagePath)
        
        // 3. استدعاء خدمات متعددة بالتوازي
        val results = coroutineScope {
            val ocrDeferred = async { cloudVisionOcr(plateCrop) }        // Cloud Vision
            val modelDeferred = async { geminiIdentifyModel(logoCrop) }   // Gemini
            val damageDeferred = async { geminiAnalyzeDamage(photo) }     // Gemini
            
            OnlineResults(
                ocr = ocrDeferred.await(),
                model = modelDeferred.await(),
                damage = damageDeferred.await()
            )
        }
        
        // 4. تحديث النتائج المحلية بصمت
        updateLocalResults(photoId, results)
        
        // 5. تحديث قاعدة المعرفة للتعلم الذاتي
        if (results.conflictsWith(photo.localResults)) {
            flagForHumanReview(photoId)
        }
        
        return Result.success()
    }
}
```
