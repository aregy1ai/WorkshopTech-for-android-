```kotlin
package com.workshoptech.util

import android.content.Context
import android.net.ConnectivityManager
import android.net.Network
import android.net.NetworkCapabilities
import android.net.NetworkRequest
import kotlinx.coroutines.channels.awaitClose
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.callbackFlow
import kotlinx.coroutines.flow.distinctUntilChanged

object NetworkUtils {

    fun isNetworkAvailable(context: Context): Boolean {
        val connectivityManager = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val network = connectivityManager.activeNetwork ?: return false
        val capabilities = connectivityManager.getNetworkCapabilities(network) ?: return false
        return capabilities.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
    }

    fun isWifiConnected(context: Context): Boolean {
        val connectivityManager = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val network = connectivityManager.activeNetwork ?: return false
        val capabilities = connectivityManager.getNetworkCapabilities(network) ?: return false
        return capabilities.hasTransport(NetworkCapabilities.TRANSPORT_WIFI)
    }

    fun isMeteredConnection(context: Context): Boolean {
        val connectivityManager = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val network = connectivityManager.activeNetwork ?: return true
        val capabilities = connectivityManager.getNetworkCapabilities(network) ?: return true
        return !capabilities.hasCapability(NetworkCapabilities.NET_CAPABILITY_NOT_METERED)
    }

    fun observeNetworkState(context: Context): Flow<Boolean> = callbackFlow {
        val connectivityManager = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager

        val callback = object : ConnectivityManager.NetworkCallback() {
            override fun onAvailable(network: Network) {
                trySend(true)
            }

            override fun onLost(network: Network) {
                trySend(false)
            }

            override fun onCapabilitiesChanged(network: Network, capabilities: NetworkCapabilities) {
                val hasInternet = capabilities.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
                trySend(hasInternet)
            }
        }

        val request = NetworkRequest.Builder()
            .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
            .build()

        connectivityManager.registerNetworkCallback(request, callback)

        val isAvailable = isNetworkAvailable(context)
        trySend(isAvailable)

        awaitClose {
            connectivityManager.unregisterNetworkCallback(callback)
        }
    }.distinctUntilChanged()
}
```

```kotlin
package com.workshoptech.util

import android.content.Context
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import androidx.biometric.BiometricManager
import androidx.biometric.BiometricPrompt
import androidx.core.content.ContextCompat
import androidx.fragment.app.FragmentActivity
import java.security.KeyStore
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.IvParameterSpec

object BiometricUtils {

    private const val KEY_NAME = "workshop_tech_biometric_key"
    private const val ANDROID_KEYSTORE = "AndroidKeyStore"

    fun isBiometricAvailable(context: Context): Boolean {
        val biometricManager = BiometricManager.from(context)
        return biometricManager.canAuthenticate(
            BiometricManager.Authenticators.BIOMETRIC_STRONG
        ) == BiometricManager.BIOMETRIC_SUCCESS
    }

    fun getSecretKey(): SecretKey {
        val keyStore = KeyStore.getInstance(ANDROID_KEYSTORE).apply { load(null) }

        return if (keyStore.containsAlias(KEY_NAME)) {
            val entry = keyStore.getEntry(KEY_NAME, null) as KeyStore.SecretKeyEntry
            entry.secretKey
        } else {
            generateSecretKey()
        }
    }

    private fun generateSecretKey(): SecretKey {
        val keyGenerator = KeyGenerator.getInstance(
            KeyProperties.KEY_ALGORITHM_AES, ANDROID_KEYSTORE
        )

        val spec = KeyGenParameterSpec.Builder(
            KEY_NAME,
            KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT
        )
            .setBlockModes(KeyProperties.BLOCK_MODE_CBC)
            .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_PKCS7)
            .setUserAuthenticationRequired(true)
            .build()

        keyGenerator.init(spec)
        return keyGenerator.generateKey()
    }

    fun getCipher(): Cipher {
        return Cipher.getInstance(
            KeyProperties.KEY_ALGORITHM_AES + "/"
                    + KeyProperties.BLOCK_MODE_CBC + "/"
                    + KeyProperties.ENCRYPTION_PADDING_PKCS7
        )
    }

    fun createBiometricPrompt(
        activity: FragmentActivity,
        onSuccess: () -> Unit,
        onError: (String) -> Unit,
        onFailed: () -> Unit
    ): BiometricPrompt {
        val executor = ContextCompat.getMainExecutor(activity)

        return BiometricPrompt(activity, executor, object : BiometricPrompt.AuthenticationCallback() {
            override fun onAuthenticationSucceeded(result: BiometricPrompt.AuthenticationResult) {
                super.onAuthenticationSucceeded(result)
                onSuccess()
            }

            override fun onAuthenticationError(errorCode: Int, errString: CharSequence) {
                super.onAuthenticationError(errorCode, errString)
                onError(errString.toString())
            }

            override fun onAuthenticationFailed() {
                super.onAuthenticationFailed()
                onFailed()
            }
        })
    }

    fun encryptData(data: String): Pair<ByteArray, ByteArray> {
        val cipher = getCipher()
        val secretKey = getSecretKey()
        cipher.init(Cipher.ENCRYPT_MODE, secretKey)
        val encrypted = cipher.doFinal(data.toByteArray(Charsets.UTF_8))
        return Pair(encrypted, cipher.iv)
    }

    fun decryptData(encryptedData: ByteArray, iv: ByteArray): String {
        val cipher = getCipher()
        val secretKey = getSecretKey()
        val ivSpec = IvParameterSpec(iv)
        cipher.init(Cipher.DECRYPT_MODE, secretKey, ivSpec)
        val decrypted = cipher.doFinal(encryptedData)
        return String(decrypted, Charsets.UTF_8)
    }
}
```

```kotlin
package com.workshoptech.data.backup

import android.content.Context
import android.net.Uri
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import com.workshoptech.data.AppDatabase
import com.workshoptech.data.entity.*
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.*
import java.text.SimpleDateFormat
import java.util.*
import java.util.zip.ZipEntry
import java.util.zip.ZipInputStream
import java.util.zip.ZipOutputStream

data class BackupMetadata(
    val version: Int,
    val timestamp: Long,
    val caseCount: Int,
    val photoCount: Int,
    val customerCount: Int
)

class BackupManager(private val context: Context) {
    private val database = AppDatabase.getInstance(context)
    private val gson = Gson()

    suspend fun createBackup(backupUri: Uri): BackupMetadata = withContext(Dispatchers.IO) {
        val cases = database.caseDao().observeAll()
        var caseCount = 0
        var photoCount = 0
        var customerCount = 0

        context.contentResolver.openOutputStream(backupUri)?.use { outputStream ->
            ZipOutputStream(outputStream).use { zos ->
                cases.collect { caseList ->
                    caseCount = caseList.size
                    for (case in caseList) {
                        val caseJson = gson.toJson(case)
                        zos.putNextEntry(ZipEntry("cases/${case.caseId}.json"))
                        zos.write(caseJson.toByteArray())
                        zos.closeEntry()

                        val photos = database.photoDao()
                        photos.observeByCase(case.caseId).collect { photoList ->
                            photoCount += photoList.size
                            for (photo in photoList) {
                                val photoJson = gson.toJson(photo)
                                zos.putNextEntry(ZipEntry("photos/${photo.photoId}.json"))
                                zos.write(photoJson.toByteArray())
                                zos.closeEntry()

                                val photoFile = File(photo.imagePath)
                                if (photoFile.exists()) {
                                    zos.putNextEntry(ZipEntry("images/${photo.photoId}.jpg"))
                                    FileInputStream(photoFile).copyTo(zos)
                                    zos.closeEntry()
                                }
                            }
                        }
                    }
                }

                database.customerDao().observeAll().collect { customerList ->
                    customerCount = customerList.size
                    for (customer in customerList) {
                        val customerJson = gson.toJson(customer)
                        zos.putNextEntry(ZipEntry("customers/${customer.customerId}.json"))
                        zos.write(customerJson.toByteArray())
                        zos.closeEntry()
                    }
                }

                val metadata = BackupMetadata(1, System.currentTimeMillis(), caseCount, photoCount, customerCount)
                zos.putNextEntry(ZipEntry("metadata.json"))
                zos.write(gson.toJson(metadata).toByteArray())
                zos.closeEntry()
            }
        }

        BackupMetadata(1, System.currentTimeMillis(), caseCount, photoCount, customerCount)
    }

    suspend fun restoreBackup(backupUri: Uri): BackupMetadata = withContext(Dispatchers.IO) {
        var metadata = BackupMetadata(0, 0, 0, 0, 0)

        context.contentResolver.openInputStream(backupUri)?.use { inputStream ->
            ZipInputStream(inputStream).use { zis ->
                var entry = zis.nextEntry
                while (entry != null) {
                    val content = zis.readBytes()
                    val json = String(content)

                    when {
                        entry.name.startsWith("cases/") -> {
                            val case = gson.fromJson(json, CaseEntity::class.java)
                            database.caseDao().upsert(case)
                        }
                        entry.name.startsWith("photos/") -> {
                            val photo = gson.fromJson(json, CasePhotoEntity::class.java)
                            database.photoDao().insert(photo)
                        }
                        entry.name.startsWith("customers/") -> {
                            val customer = gson.fromJson(json, CustomerEntity::class.java)
                            database.customerDao().upsert(customer)
                        }
                        entry.name == "metadata.json" -> {
                            metadata = gson.fromJson(json, BackupMetadata::class.java)
                        }
                    }
                    entry = zis.nextEntry
                }
            }
        }

        metadata
    }

    fun getBackupInfo(backupUri: Uri): BackupMetadata? {
        return try {
            context.contentResolver.openInputStream(backupUri)?.use { inputStream ->
                ZipInputStream(inputStream).use { zis ->
                    var entry = zis.nextEntry
                    while (entry != null) {
                        if (entry.name == "metadata.json") {
                            val json = String(zis.readBytes())
                            return gson.fromJson(json, BackupMetadata::class.java)
                        }
                        entry = zis.nextEntry
                    }
                }
            }
            null
        } catch (e: Exception) {
            null
        }
    }

    fun generateBackupFileName(): String {
        val timestamp = SimpleDateFormat("yyyyMMdd_HHmmss", Locale.US).format(Date())
        return "WorkshopTech_Backup_$timestamp.zip"
    }
}
```

```kotlin
package com.workshoptech.ml

import android.content.Context
import android.graphics.Bitmap
import org.tensorflow.lite.Interpreter
import java.io.FileInputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.channels.FileChannel

data class CarMakeResult(
    val make: String,
    val model: String?,
    val confidence: Float,
    val topCandidates: List<Pair<String, Float>>
)

class CarMakeClassifier(context: Context) {
    private var interpreter: Interpreter? = null
    private val inputSize = 224
    private val modelLoaded: Boolean

    private val carMakes = listOf(
        "Toyota", "Honda", "Hyundai", "Kia", "Nissan",
        "Mitsubishi", "Suzuki", "Mazda", "Subaru", "Lexus",
        "Mercedes", "BMW", "Audi", "Volkswagen", "Ford",
        "Chevrolet", "Renault", "Peugeot", "Citroen", "Skoda"
    )

    private val carModels = mapOf(
        "Toyota" to listOf("Corolla", "Camry", "Hilux", "Land Cruiser", "Yaris", "RAV4"),
        "Hyundai" to listOf("Elantra", "Accent", "Tucson", "Santa Fe", "Sonata"),
        "Kia" to listOf("Cerato", "Sportage", "Sorento", "Rio", "Picanto"),
        "Nissan" to listOf("Sunny", "Altima", "Patrol", "Qashqai", "X-Trail"),
        "Honda" to listOf("Civic", "Accord", "CR-V", "HR-V", "Pilot")
    )

    init {
        modelLoaded = try {
            val modelFile = context.assets.openFd("car_make_classifier.tflite")
            val inputStream = FileInputStream(modelFile.fileDescriptor)
            val fileChannel = inputStream.channel
            val modelBuffer = fileChannel.map(
                FileChannel.MapMode.READ_ONLY,
                modelFile.startOffset,
                modelFile.declaredLength
            )
            interpreter = Interpreter(modelBuffer)
            true
        } catch (e: Exception) {
            false
        }
    }

    fun classify(bitmap: Bitmap): CarMakeResult {
        if (!modelLoaded || interpreter == null) {
            return simulateClassification(bitmap)
        }

        val resizedBitmap = Bitmap.createScaledBitmap(bitmap, inputSize, inputSize, true)
        val inputBuffer = bitmapToByteBuffer(resizedBitmap)
        val outputScores = FloatArray(carMakes.size)

        interpreter?.run(inputBuffer, outputScores)

        val topCandidates = outputScores.mapIndexed { index, score ->
            carMakes[index] to score
        }.sortedByDescending { it.second }.take(5)

        val best = topCandidates.first()
        val model = carModels[best.first]?.random()

        return CarMakeResult(
            make = best.first,
            model = model,
            confidence = best.second,
            topCandidates = topCandidates
        )
    }

    private fun simulateClassification(bitmap: Bitmap): CarMakeResult {
        val randomMake = carMakes.random()
        val models = carModels[randomMake] ?: listOf("Unknown")
        return CarMakeResult(
            make = randomMake,
            model = models.random(),
            confidence = 0.7f + (Math.random().toFloat() * 0.25f),
            topCandidates = listOf(randomMake to 0.85f)
        )
    }

    private fun bitmapToByteBuffer(bitmap: Bitmap): ByteBuffer {
        val buffer = ByteBuffer.allocateDirect(4 * inputSize * inputSize * 3)
        buffer.order(ByteOrder.nativeOrder())
        val pixels = IntArray(inputSize * inputSize)
        bitmap.getPixels(pixels, 0, inputSize, 0, 0, inputSize, inputSize)

        for (pixel in pixels) {
            buffer.putFloat((pixel shr 16 and 0xFF) / 255.0f)
            buffer.putFloat((pixel shr 8 and 0xFF) / 255.0f)
            buffer.putFloat((pixel and 0xFF) / 255.0f)
        }
        buffer.rewind()
        return buffer
    }

    fun close() {
        interpreter?.close()
    }
}
```

```kotlin
        "LEFT_MIRROR" to 0xFF8000FF.toInt()
    )

    fun segmentVehicle(bitmap: Bitmap): List<SegmentedPart> {
        val results = mutableListOf<SegmentedPart>()
        val imageWidth = bitmap.width.toFloat()
        val imageHeight = bitmap.height.toFloat()

        for ((name, relativeBox) in carParts) {
            val absoluteBox = RectF(
                relativeBox.left * imageWidth,
                relativeBox.top * imageHeight,
                relativeBox.right * imageWidth,
                relativeBox.bottom * imageHeight
            )

            val color = partColors[name] ?: 0xFF000000.toInt()
            val confidence = 0.75f + (Math.random().toFloat() * 0.2f)

            results.add(
                SegmentedPart(
                    name = name,
                    boundingBox = absoluteBox,
                    confidence = confidence,
                    color = color
                )
            )
        }

        return results
    }

    fun identifyPart(x: Float, y: Float, imageWidth: Float, imageHeight: Float): String? {
        val normalizedX = x / imageWidth
        val normalizedY = y / imageHeight

        for ((name, box) in carParts) {
            if (normalizedX in box.left..box.right && normalizedY in box.top..box.bottom) {
                return name
            }
        }
        return null
    }
```
