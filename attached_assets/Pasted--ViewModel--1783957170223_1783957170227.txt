data class CustomerState(
    val query: String = "",
    val items: List<CustomerEntity> = emptyList(),
    val isLoading: Boolean = false,
    val selectedCustomer: CustomerEntity? = null
)

class CustomerViewModel(
    private val repository: WorkshopRepository
) : ViewModel() {
    private val queryFlow = MutableStateFlow("")
    private val _state = MutableStateFlow(CustomerState())
    val state: StateFlow<CustomerState> = _state.asStateFlow()

    init {
        viewModelScope.launch {
            queryFlow
                .debounce(300)
                .flatMapLatest { q ->
                    repository.observeCustomers(q.ifBlank { null })
                }
                .collect { items ->
                    _state.value = _state.value.copy(
                        items = items,
                        query = queryFlow.value
                    )
                }
        }
    }

    fun setQuery(q: String) {
        queryFlow.value = q
    }

    fun selectCustomer(customer: CustomerEntity) {
        _state.value = _state.value.copy(selectedCustomer = customer)
    }

    fun clearSelection() {
        _state.value = _state.value.copy(selectedCustomer = null)
    }
}

@Composable
fun InspectionStatusCard(
    modifier: Modifier = Modifier,
    label: String,
    title: String,
    status: String?
) {
    Card(
        modifier = modifier,
        colors = CardDefaults.cardColors(
            containerColor = when (status) {
                "PASSED" -> MaterialTheme.colorScheme.tertiaryContainer
                "PASSED_WITH_NOTES" -> MaterialTheme.colorScheme.secondaryContainer
                "FAILED" -> MaterialTheme.colorScheme.errorContainer
                else -> MaterialTheme.colorScheme.surface
            }
        )
    ) {
        Column(
            modifier = Modifier.padding(8.dp),
            horizontalAlignment = androidx.compose.ui.Alignment.CenterHorizontally
        ) {
            Text(label, fontWeight = FontWeight.Bold)
            Text(title, style = MaterialTheme.typography.bodySmall)
            Icon(
                imageVector = when (status) {
                    "PASSED" -> Icons.Default.CheckCircle
                    "PASSED_WITH_NOTES" -> Icons.Default.Warning
                    "FAILED" -> Icons.Default.Cancel
                    else -> Icons.Default.Help
                },
                contentDescription = null,
                modifier = Modifier.size(20.dp),
                tint = when (status) {
                    "PASSED" -> MaterialTheme.colorScheme.tertiary
                    "PASSED_WITH_NOTES" -> MaterialTheme.colorScheme.secondary
                    "FAILED" -> MaterialTheme.colorScheme.error
                    else -> MaterialTheme.colorScheme.onSurface.copy(alpha = 0.4f)
                }
            )
        }
    }
}
