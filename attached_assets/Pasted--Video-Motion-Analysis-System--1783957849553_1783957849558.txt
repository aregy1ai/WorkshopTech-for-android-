```kotlin
package com.workshoptech.video

import android.content.Context
import android.graphics.Bitmap
import android.media.MediaMetadataRetriever
import java.io.File
import java.io.FileOutputStream

data class ExtractedFrame(
    val bitmap: Bitmap,
    val timestampMs: Long,
    val frameNumber: Int
)

class FrameExtractor(private val context: Context) {

    private val retriever = MediaMetadataRetriever()

    fun extractFrames(
        videoPath: String,
        intervalMs: Long = 500L,
        maxFrames: Int = 120
    ): List<ExtractedFrame> {
        val frames = mutableListOf<ExtractedFrame>()

        try {
            retriever.setDataSource(videoPath)
            val durationStr = retriever.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION)
            val durationMs = durationStr?.toLongOrNull() ?: return frames

            val frameInterval = (durationMs / maxFrames).coerceAtLeast(intervalMs)
            var currentMs = 0L
            var frameNumber = 0

            while (currentMs < durationMs && frameNumber < maxFrames) {
                val bitmap = retriever.getFrameAtTime(
                    currentMs * 1000,
                    MediaMetadataRetriever.OPTION_CLOSEST_SYNC
                )

                if (bitmap != null) {
                    frames.add(
                        ExtractedFrame(
                            bitmap = bitmap,
                            timestampMs = currentMs,
                            frameNumber = frameNumber
                        )
                    )
                    frameNumber++
                }

                currentMs += frameInterval
            }
        } catch (e: Exception) {
            e.printStackTrace()
        } finally {
            try { retriever.release() } catch (_: Exception) {}
        }

        return frames
    }

    fun saveFrame(bitmap: Bitmap, outputFile: File, quality: Int = 90): Boolean {
        return try {
            FileOutputStream(outputFile).use { out ->
                bitmap.compress(Bitmap.CompressFormat.JPEG, quality, out)
            }
            true
        } catch (e: Exception) {
            false
        }
    }

    fun extractKeyFrames(videoPath: String, maxKeyFrames: Int = 10): List<ExtractedFrame> {
        val allFrames = extractFrames(videoPath, intervalMs = 200L, maxFrames = 300)
        if (allFrames.size <= maxKeyFrames) return allFrames

        val keyFrames = mutableListOf<ExtractedFrame>()
        keyFrames.add(allFrames.first())
        keyFrames.add(allFrames.last())

        val step = (allFrames.size - 1).toFloat() / (maxKeyFrames - 1)
        for (i in 1 until maxKeyFrames - 1) {
            val index = (i * step).toInt().coerceIn(0, allFrames.size - 1)
            keyFrames.add(allFrames[index])
        }

        return keyFrames
    }
}
```

```kotlin
package com.workshoptech.video

import android.graphics.Bitmap
import android.graphics.Color
import kotlin.math.abs
import kotlin.math.sqrt

data class MotionPoint(
    val x: Float,
    val y: Float,
    val timestampMs: Long
)

data class MotionTrack(
    val points: List<MotionPoint>,
    val averageVelocity: Float,
    val maxVelocity: Float,
    val totalDistance: Float,
    val motionType: String
)

data class MotionAnalysisResult(
    val tracks: List<MotionTrack>,
    val stationaryObjects: Int,
    val movingObjects: Int,
    val averageMotionSpeed: Float,
    val motionPattern: String
)

class MotionTracker {

    private val minFeatureDistance = 10.0
    private val maxTrackingPoints = 50
    private var previousPoints = mutableListOf<MotionPoint>()

    fun trackMotion(
        currentFrame: Bitmap,
        previousFrame: Bitmap?,
        timestampMs: Long
    ): MotionAnalysisResult {
        if (previousFrame == null) {
            initializeTracking(currentFrame, timestampMs)
            return MotionAnalysisResult(
                tracks = emptyList(),
                stationaryObjects = 0,
                movingObjects = 0,
                averageMotionSpeed = 0f,
                motionPattern = "INITIALIZING"
            )
        }

        val currentPoints = extractFeaturePoints(currentFrame)
        val tracks = matchPoints(previousPoints, currentPoints, timestampMs)
        val velocities = calculateVelocities(tracks)

        previousPoints = currentPoints.toMutableList()

        val movingObjects = tracks.count { it.averageVelocity > 0.5f }
        val stationaryObjects = tracks.size - movingObjects
        val avgSpeed = if (velocities.isNotEmpty()) velocities.average().toFloat() else 0f

        val motionPattern = when {
            avgSpeed > 10f -> "FAST_MOTION"
            avgSpeed > 2f -> "NORMAL_MOTION"
            avgSpeed > 0.5f -> "SLOW_MOTION"
            else -> "STATIONARY"
        }

        return MotionAnalysisResult(
            tracks = tracks,
            stationaryObjects = stationaryObjects,
            movingObjects = movingObjects,
            averageMotionSpeed = avgSpeed,
            motionPattern = motionPattern
        )
    }

    private fun extractFeaturePoints(bitmap: Bitmap): List<MotionPoint> {
        val points = mutableListOf<MotionPoint>()
        val width = bitmap.width
        val height = bitmap.height
        val pixels = IntArray(width * height)
        bitmap.getPixels(pixels, 0, width, 0, 0, width, height)

        val stepX = width / 10
        val stepY = height / 10

        for (y in stepY until height - stepY step stepY) {
            for (x in stepX until width - stepX step stepX) {
                val idx = y * width + x
                val current = Color.green(pixels[idx])
                val right = Color.green(pixels[idx + 1])
                val bottom = Color.green(pixels[idx + width])

                val gradient = abs(current - right) + abs(current - bottom)
                if (gradient > 50) {
                    points.add(
                        MotionPoint(
                            x = x.toFloat() / width,
                            y = y.toFloat() / height,
                            timestampMs = System.currentTimeMillis()
                        )
                    )
                }
            }
        }

        return points.take(maxTrackingPoints)
    }

    private fun initializeTracking(bitmap: Bitmap, timestampMs: Long) {
        previousPoints = extractFeaturePoints(bitmap).map {
            it.copy(timestampMs = timestampMs)
        }.toMutableList()
    }

    private fun matchPoints(
        previous: List<MotionPoint>,
        current: List<MotionPoint>,
        timestampMs: Long
    ): List<MotionTrack> {
        val tracks = mutableListOf<MotionTrack>()

        for (prevPoint in previous) {
            var bestMatch: MotionPoint? = null
            var bestDistance = Double.MAX_VALUE

            for (currPoint in current) {
                val dx = prevPoint.x - currPoint.x
                val dy = prevPoint.y - currPoint.y
                val distance = sqrt(dx * dx + dy * dy)

                if (distance < bestDistance && distance < 0.1) {
                    bestDistance = distance
                    bestMatch = currPoint
                }
            }

            if (bestMatch != null) {
                val dt = (timestampMs - prevPoint.timestampMs).coerceAtLeast(1)
                val velocity = bestDistance.toFloat() / (dt / 1000f)
                val track = MotionTrack(
                    points = listOf(prevPoint, bestMatch),
                    averageVelocity = velocity,
                    maxVelocity = velocity,
                    totalDistance = bestDistance.toFloat(),
                    motionType = if (velocity > 1f) "MOVING" else "STATIC"
                )
                tracks.add(track)
            }
        }

        return tracks
    }

    private fun calculateVelocities(tracks: List<MotionTrack>): List<Float> {
        return tracks.filter { it.motionType == "MOVING" }.map { it.averageVelocity }
    }

    fun reset() {
        previousPoints.clear()
    }
}
```

```kotlin
package com.workshoptech.video

import android.content.Context
import android.graphics.Bitmap
import com.workshoptech.ml.DamageAnalyzer
import com.workshoptech.ml.QualityFilter
import com.workshoptech.domain.model.DamageReport
import java.io.File
import java.util.UUID

data class VideoAnalysisResult(
    val videoId: String,
    val totalFrames: Int,
    val analyzedFrames: Int,
    val framesWithDamage: Int,
    val damageReports: List<FrameDamageReport>,
    val motionAnalysis: MotionAnalysisResult?,
    val surfaceQuality: SurfaceQualityReport?,
    val alignmentIssues: List<AlignmentIssue>?,
    val durationMs: Long,
    val overallAssessment: String
)

data class FrameDamageReport(
    val frameNumber: Int,
    val timestampMs: Long,
    val damageReport: DamageReport,
    val framePath: String?
)

data class SurfaceQualityReport(
    val overallScore: Float,
    val reflectionUniformity: Float,
    val orangePeelScore: Float,
    val defectCount: Int,
    val affectedAreas: List<String>
)

data class AlignmentIssue(
    val partName: String,
    val gapWidth: Float,
    val isAligned: Boolean,
    val severity: String,
    val description: String
)

class VideoAnalyzer(private val context: Context) {

    private val frameExtractor = FrameExtractor(context)
    private val damageAnalyzer = DamageAnalyzer(context)
    private val motionTracker = MotionTracker()
    private val surfaceAnalyzer = SurfaceAnalyzer()
    private val alignmentDetector = AlignmentDetector()

    suspend fun analyzeVideo(
        videoPath: String,
        videoId: String,
        onProgress: (Float) -> Unit = {}
    ): VideoAnalysisResult {
        val frames = frameExtractor.extractFrames(videoPath, intervalMs = 500L)
        val damageReports = mutableListOf<FrameDamageReport>()
        var previousFrame: Bitmap? = null
        var totalMotionAnalysis: MotionAnalysisResult? = null

        val framesDir = File(context.filesDir, "video_frames/$videoId")
        if (!framesDir.exists()) framesDir.mkdirs()

        for ((index, frame) in frames.withIndex()) {
            val progress = (index + 1).toFloat() / frames.size
            onProgress(progress)

            val qualityResult = QualityFilter.analyze(frame.bitmap)
            if (!qualityResult.isAcceptable) continue

            val damageReport = damageAnalyzer.analyze(frame.bitmap)
            var framePath: String? = null

            if (damageReport.regions.isNotEmpty()) {
                val frameFile = File(framesDir, "frame_${frame.frameNumber}.jpg")
                frameExtractor.saveFrame(frame.bitmap, frameFile)
                framePath = frameFile.absolutePath
            }

            damageReports.add(
                FrameDamageReport(
                    frameNumber = frame.frameNumber,
                    timestampMs = frame.timestampMs,
                    damageReport = damageReport,
                    framePath = framePath
                )
            )

            val motionResult = motionTracker.trackMotion(
                frame.bitmap, previousFrame, frame.timestampMs
            )
            if (motionResult.motionPattern != "INITIALIZING") {
                totalMotionAnalysis = motionResult
            }
            previousFrame = frame.bitmap
        }

        val surfaceQuality = if (frames.isNotEmpty()) {
            surfaceAnalyzer.analyzeSurfaceQuality(frames.map { it.bitmap })
        } else null

        val alignmentIssues = if (frames.isNotEmpty()) {
            alignmentDetector.detectAlignmentIssues(frames.map { it.bitmap })
        } else null

        val framesWithDamage = damageReports.count { it.damageReport.regions.isNotEmpty() }
        val overallAssessment = when {
            framesWithDamage > frames.size * 0.5 -> "HEAVILY_DAMAGED"
            framesWithDamage > frames.size * 0.2 -> "MODERATELY_DAMAGED"
            framesWithDamage > 0 -> "MINOR_DAMAGE"
            else -> "NO_DAMAGE_DETECTED"
        }

        return VideoAnalysisResult(
            videoId = videoId,
            totalFrames = frames.size,
            analyzedFrames = damageReports.size,
            framesWithDamage = framesWithDamage,
            damageReports = damageReports,
            motionAnalysis = totalMotionAnalysis,
            surfaceQuality = surfaceQuality,
            alignmentIssues = alignmentIssues,
            durationMs = frames.lastOrNull()?.timestampMs ?: 0L,
            overallAssessment = overallAssessment
        )
    }
}
```

119. video/SurfaceAnalyzer.kt

```kotlin
package com.workshoptech.video

import android.graphics.Bitmap
import android.graphics.Color
import kotlin.math.abs
import kotlin.math.sqrt

data class SurfaceRegion(
    val centerX: Float,
    val centerY: Float,
    val radius: Float,
    val reflectionIntensity: Float,
    val uniformityScore: Float
)

class SurfaceAnalyzer {

    fun analyzeSurfaceQuality(frames: List<Bitmap>): SurfaceQualityReport {
        if (frames.isEmpty()) {
            return SurfaceQualityReport(0f, 0f, 0f, 0, emptyList())
        }

        val allDefects = mutableListOf<SurfaceRegion>()
        var totalUniformity = 0f
        var totalOrangePeel = 0f

        for (frame in frames) {
            val regions = analyzeFrameSurface(frame)
            allDefects.addAll(regions)

            val uniformity = calculateUniformity(frame)
            val orangePeel = detectOrangePeel(frame)

            totalUniformity += uniformity
            totalOrangePeel += orangePeel
        }

        val avgUniformity = totalUniformity / frames.size
        val avgOrangePeel = totalOrangePeel / frames.size
        val defectCount = allDefects.size

        val overallScore = (avgUniformity * 0.4f + (1f - avgOrangePeel) * 0.4f +
                (1f - (defectCount.toFloat() / (frames.size * 10))) * 0.2f) * 100f

        val affectedAreas = identifyAffectedAreas(allDefects)

        return SurfaceQualityReport(
            overallScore = overallScore.coerceIn(0f, 100f),
            reflectionUniformity = avgUniformity * 100f,
            orangePeelScore = avgOrangePeel * 100f,
            defectCount = defectCount,
            affectedAreas = affectedAreas
        )
    }

    private fun analyzeFrameSurface(bitmap: Bitmap): List<SurfaceRegion> {
        val regions = mutableListOf<SurfaceRegion>()
        val width = bitmap.width
        val height = bitmap.height
        val pixels = IntArray(width * height)
        bitmap.getPixels(pixels, 0, width, 0, 0, width, height)

        val blockSize = 20
        for (y in 0 until height step blockSize) {
            for (x in 0 until width step blockSize) {
                var sumIntensity = 0f
                var sumSquaredDiff = 0f
                var count = 0

                for (by in 0 until blockSize) {
                    for (bx in 0 until blockSize) {
                        val px = x + bx
                        val py = y + by
                        if (px < width && py < height) {
                            val pixel = pixels[py * width + px]
                            val intensity = (Color.red(pixel) + Color.green(pixel) + Color.blue(pixel)) / 3f
                            sumIntensity += intensity
                            count++
                        }
                    }
                }

                if (count > 0) {
                    val avgIntensity = sumIntensity / count

                    for (by in 0 until blockSize) {
                        for (bx in 0 until blockSize) {
                            val px = x + bx
                            val py = y + by
                            if (px < width && py < height) {
                                val pixel = pixels[py * width + px]
                                val intensity = (Color.red(pixel) + Color.green(pixel) + Color.blue(pixel)) / 3f
                                sumSquaredDiff += (intensity - avgIntensity) * (intensity - avgIntensity)
                            }
                        }
                    }

                    val variance = sumSquaredDiff / count
                    val uniformity = 1f - (sqrt(variance) / 128f).coerceIn(0f, 1f)

                    if (uniformity < 0.7f) {
                        regions.add(
                            SurfaceRegion(
                                centerX = (x + blockSize / 2).toFloat() / width,
                                centerY = (y + blockSize / 2).toFloat() / height,
                                radius = blockSize.toFloat() / minOf(width, height),
                                reflectionIntensity = avgIntensity / 255f,
                                uniformityScore = uniformity
                            )
                        )
                    }
                }
            }
        }

        return regions
    }

    private fun calculateUniformity(bitmap: Bitmap): Float {
        val width = bitmap.width
        val height = bitmap.height
        val pixels = IntArray(width * height)
        bitmap.getPixels(pixels, 0, width, 0, 0, width, height)

        val intensities = mutableListOf<Float>()
        for (pixel in pixels) {
            val intensity = (Color.red(pixel) + Color.green(pixel) + Color.blue(pixel)) / 3f
            intensities.add(intensity)
        }

        val mean = intensities.average().toFloat()
        val variance = intensities.map { (it - mean) * (it - mean) }.average().toFloat()
        val stdDev = sqrt(variance)

        return (1f - (stdDev / 128f)).coerceIn(0f, 1f)
    }

    private fun detectOrangePeel(bitmap: Bitmap): Float {
        val width = bitmap.width
        val height = bitmap.height
        val pixels = IntArray(width * height)
        bitmap.getPixels(pixels, 0, width, 0, 0, width, height)

        var gradientCount = 0
        var totalGradients = 0

        for (y in 1 until height - 1) {
            for (x in 1 until width - 1) {
                val idx = y * width + x
                val current = Color.green(pixels[idx])
                val right = Color.green(pixels[idx + 1])
                val bottom = Color.green(pixels[idx + width])

                if (abs(current - right) > 15 || abs(current - bottom) > 15) {
                    gradientCount++
                }
                totalGradients++
            }
        }

        return (gradientCount.toFloat() / totalGradients).coerceIn(0f, 1f)
    }

    private fun identifyAffectedAreas(regions: List<SurfaceRegion>): List<String> {
        val areaMap = mapOf(
            "HOOD" to { r: SurfaceRegion -> r.centerY < 0.35f && r.centerX in 0.15f..0.85f },
            "RIGHT_DOOR" to { r: SurfaceRegion -> r.centerX > 0.65f && r.centerY in 0.35f..0.7f },
            "LEFT_DOOR" to { r: SurfaceRegion -> r.centerX < 0.35f && r.centerY in 0.35f..0.7f },
            "TRUNK" to { r: SurfaceRegion -> r.centerY > 0.7f && r.centerX in 0.2f..0.8f },
            "ROOF" to { r: SurfaceRegion -> r.centerY < 0.15f && r.centerX in 0.15f..0.85f }
        )

        return areaMap.filter { (_, check) ->
            regions.any { check(it) }
        }.keys.toList()
    }
}
```

```kotlin
package com.workshoptech.video

import android.graphics.Bitmap
import android.graphics.Color
import kotlin.math.abs

data class GapMeasurement(
    val startX: Float,
    val startY: Float,
    val endX: Float,
    val endY: Float,
    val width: Float,
    val position: String
)

class AlignmentDetector {

    private val minGapThreshold = 2f
    private val maxGapThreshold = 8f
    private val targetGapWidth = 4.5f

    fun detectAlignmentIssues(frames: List<Bitmap>): List<AlignmentIssue> {
        val issues = mutableListOf<AlignmentIssue>()
        val allGaps = mutableListOf<GapMeasurement>()

        for (frame in frames) {
            val gaps = detectGaps(frame)
            allGaps.addAll(gaps)
        }

        val gapGroups = allGaps.groupBy { it.position }

        for ((position, gaps) in gapGroups) {
            val avgWidth = gaps.map { it.width }.average().toFloat()
            val isAligned = abs(avgWidth - targetGapWidth) < 1.5f
            val severity = when {
                abs(avgWidth - targetGapWidth) > 3f -> "HIGH"
                abs(avgWidth - targetGapWidth) > 1.5f -> "MEDIUM"
                else -> "LOW"
            }

            val description = when {
                avgWidth < minGapThreshold -> "فجوة ضيقة جداً - قد تحتك القطع ببعضها"
                avgWidth > maxGapThreshold -> "فجوة واسعة جداً - تحتاج إعادة محاذاة"
                !isAligned -> "المحاذاة غير متساوية"
                else -> "المحاذاة مقبولة"
            }

            issues.add(
                AlignmentIssue(
                    partName = getPartName(position),
                    gapWidth = avgWidth,
                    isAligned = isAligned,
                    severity = severity,
                    description = description
                )
            )
        }

        return issues
    }

    private fun detectGaps(bitmap: Bitmap): List<GapMeasurement> {
        val gaps = mutableListOf<GapMeasurement>()
        val width = bitmap.width
        val height = bitmap.height
        val pixels = IntArray(width * height)
        bitmap.getPixels(pixels, 0, width, 0, 0, width, height)

        val scanLines = listOf(
            0.25f to "HOOD_GAP",
            0.45f to "DOOR_GAP_FRONT",
            0.55f to "DOOR_GAP_REAR",
            0.75f to "TRUNK_GAP",
            0.35f to "FENDER_GAP_FRONT",
            0.65f to "FENDER_GAP_REAR"
        )

        for ((relativeY, position) in scanLines) {
            val y = (relativeY * height).toInt()
            var gapStart = -1
            var inGap = false

            for (x in 0 until width) {
                val pixel = pixels[y * width + x]
                val intensity = (Color.red(pixel) + Color.green(pixel) + Color.blue(pixel)) / 3

                if (intensity < 60 && !inGap) {
                    gapStart = x
                    inGap = true
                } else if (intensity >= 60 && inGap) {
                    val gapWidth = (x - gapStart).toFloat() * 100f / width
                    if (gapWidth in 1f..20f) {
                        gaps.add(
                            GapMeasurement(
                                startX = gapStart.toFloat() / width,
                                startY = relativeY,
                                endX = x.toFloat() / width,
                                endY = relativeY,
                                width = gapWidth,
                                position = position
                            )
                        )
                    }
                    inGap = false
                }
            }
        }

        return gaps
    }

    private fun getPartName(position: String): String {
        return when (position) {
            "HOOD_GAP" -> "غطاء المحرك"
            "DOOR_GAP_FRONT" -> "الباب الأمامي"
            "DOOR_GAP_REAR" -> "الباب الخلفي"
            "TRUNK_GAP" -> "صندوق خلفي"
            "FENDER_GAP_FRONT" -> "رفرف أمامي"
            "FENDER_GAP_REAR" -> "رفرف خلفي"
            else -> position
        }
    }
```

```kotlin
package com.workshoptech.video

import android.graphics.Bitmap
import com.workshoptech.domain.model.DamageRegion
import com.workshoptech.ml.DamageAnalyzer

data class Damage3DReport(
    val damageId: String,
    val positions: List<TrackedPosition>,
    val totalMovement: Float,
    val damageType: String,
    val affectedParts: List<String>,
    val severityProgression: String
)

data class TrackedPosition(
    val frameNumber: Int,
    val x: Float,
    val y: Float,
    val width: Float,
    val height: Float,
    val confidence: Float
)

class DamageTracker3D {

    private val damageAnalyzer = DamageAnalyzer(null)
    private val trackedDamages = mutableMapOf<String, MutableList<TrackedPosition>>()
    private var frameCounter = 0

    fun trackDamageAcrossFrames(
        frames: List<Bitmap>,
        onProgress: (Float) -> Unit = {}
    ): List<Damage3DReport> {
        trackedDamages.clear()
        frameCounter = 0

        for ((index, frame) in frames.withIndex()) {
            onProgress((index + 1).toFloat() / frames.size)
            frameCounter = index
            analyzeAndTrack(frame)
        }

        return generateReports()
    }

    private fun analyzeAndTrack(bitmap: Bitmap) {
        val report = damageAnalyzer.analyze(bitmap)

        for (region in report.regions) {
            val damageId = findMatchingDamage(region)
            val position = TrackedPosition(
                frameNumber = frameCounter,
                x = region.boundingBox.left,
                y = region.boundingBox.top,
                width = region.boundingBox.right - region.boundingBox.left,
                height = region.boundingBox.bottom - region.boundingBox.top,
                confidence = region.confidence
            )

            if (damageId != null) {
                trackedDamages[damageId]?.add(position)
            } else {
                val newId = "DMG_${trackedDamages.size}_${System.currentTimeMillis()}"
                trackedDamages[newId] = mutableListOf(position)
            }
        }
    }

    private fun findMatchingDamage(region: DamageRegion): String? {
        val threshold = 0.3f
        val cx = (region.boundingBox.left + region.boundingBox.right) / 2f
        val cy = (region.boundingBox.top + region.boundingBox.bottom) / 2f

        for ((id, positions) in trackedDamages) {
            val lastPos = positions.lastOrNull() ?: continue
            val lastCx = lastPos.x + lastPos.width / 2f
            val lastCy = lastPos.y + lastPos.height / 2f

            val distance = kotlin.math.sqrt(
                (cx - lastCx) * (cx - lastCx) + (cy - lastCy) * (cy - lastCy)
            )

            if (distance < threshold) return id
        }

        return null
    }

    private fun generateReports(): List<Damage3DReport> {
        return trackedDamages.map { (id, positions) ->
            val firstPos = positions.first()
            val lastPos = positions.last()
            val totalMovement = kotlin.math.sqrt(
                (lastPos.x - firstPos.x) * (lastPos.x - firstPos.x) +
                (lastPos.y - firstPos.y) * (lastPos.y - firstPos.y)
            )

            val confidences = positions.map { it.confidence }
            val severityProgression = when {
                confidences.average() > 0.8f -> "STABLE_HIGH"
                confidences.last() > confidences.first() -> "INCREASING"
                confidences.last() < confidences.first() -> "DECREASING"
                else -> "STABLE"
            }

            Damage3DReport(
                damageId = id,
                positions = positions,
                totalMovement = totalMovement,
                damageType = "TRACKED_DAMAGE",
                affectedParts = listOf("DETECTED"),
                severityProgression = severityProgression
            )
        }
    }

    fun reset() {
        trackedDamages.clear()
        frameCounter = 0
    }
}
```

```kotlin
package com.workshoptech.video

import android.graphics.Bitmap
import android.graphics.Color

data class FlowVector(
    val startX: Float,
    val startY: Float,
    val endX: Float,
    val endY: Float,
    val magnitude: Float,
    val angle: Float
)

data class OpticalFlowResult(
    val vectors: List<FlowVector>,
    val averageMagnitude: Float,
    val dominantDirection: Float,
    val flowPattern: String,
    val anomalyRegions: List<Pair<Float, Float>>
)

class OpticalFlowAnalyzer {

    private val blockSize = 16
    private val searchRadius = 8

    fun analyzeOpticalFlow(
        previousFrame: Bitmap,
        currentFrame: Bitmap
    ): OpticalFlowResult {
        val vectors = mutableListOf<FlowVector>()
        val prevWidth = previousFrame.width
        val prevHeight = previousFrame.height
        val currWidth = currentFrame.width
        val currHeight = currentFrame.height

        val prevPixels = IntArray(prevWidth * prevHeight)
        val currPixels = IntArray(currWidth * currHeight)
        previousFrame.getPixels(prevPixels, 0, prevWidth, 0, 0, prevWidth, prevHeight)
        currentFrame.getPixels(currPixels, 0, currWidth, 0, 0, currWidth, currHeight)

        for (y in 0 until prevHeight step blockSize) {
            for (x in 0 until prevWidth step blockSize) {
                val flow = findBestMatch(
                    prevPixels, currPixels,
                    x, y, prevWidth, prevHeight,
                    currWidth, currHeight
                )

                if (flow != null) {
                    vectors.add(flow)
                }
            }
        }

        val avgMagnitude = if (vectors.isNotEmpty()) {
            vectors.map { it.magnitude }.average().toFloat()
        } else 0f

        val dominantDir = if (vectors.isNotEmpty()) {
            vectors.map { it.angle }.average().toFloat()
        } else 0f

        val flowPattern = when {
            avgMagnitude < 0.5f -> "MINIMAL_MOTION"
            avgMagnitude < 2f -> "SLOW_ROTATION"
            avgMagnitude < 5f -> "NORMAL_MOTION"
            else -> "FAST_MOTION"
        }

        val anomalies = detectFlowAnomalies(vectors, avgMagnitude)

        return OpticalFlowResult(
            vectors = vectors,
            averageMagnitude = avgMagnitude,
            dominantDirection = dominantDir,
            flowPattern = flowPattern,
            anomalyRegions = anomalies
        )
    }

    private fun findBestMatch(
        prevPixels: IntArray,
        currPixels: IntArray,
        startX: Int,
        startY: Int,
        prevWidth: Int,
        prevHeight: Int,
        currWidth: Int,
        currHeight: Int
    ): FlowVector? {
        var bestMatch = Float.MAX_VALUE
        var bestDx = 0
        var bestDy = 0

        val prevBlock = extractBlock(prevPixels, startX, startY, blockSize, prevWidth, prevHeight)
        if (prevBlock.isEmpty()) return null

        for (dy in -searchRadius..searchRadius) {
            for (dx in -searchRadius..searchRadius) {
                val currX = (startX + dx).coerceIn(0, currWidth - blockSize)
                val currY = (startY + dy).coerceIn(0, currHeight - blockSize)
                val currBlock = extractBlock(currPixels, currX, currY, blockSize, currWidth, currHeight)

                if (currBlock.size == prevBlock.size) {
                    val sad = sumAbsoluteDifference(prevBlock, currBlock)
                    if (sad < bestMatch) {
                        bestMatch = sad
                        bestDx = dx
                        bestDy = dy
                    }
                }
            }
        }

        if (bestMatch > 5000f) return null

        val magnitude = kotlin.math.sqrt((bestDx * bestDx + bestDy * bestDy).toFloat())
        val angle = Math.toDegrees(
            kotlin.math.atan2(bestDy.toDouble(), bestDx.toDouble())
        ).toFloat()

        return FlowVector(
            startX = startX.toFloat() / prevWidth,
            startY = startY.toFloat() / prevHeight,
            endX = (startX + bestDx).toFloat() / prevWidth,
            endY = (startY + bestDy).toFloat() / prevHeight,
            magnitude = magnitude,
            angle = angle
        )
    }

    private fun extractBlock(
        pixels: IntArray,
        x: Int,
        y: Int,
        size: Int,
        width: Int,
        height: Int
    ): IntArray {
        val block = mutableListOf<Int>()
        for (by in 0 until size) {
            for (bx in 0 until size) {
                val px = x + bx
                val py = y + by
                if (px < width && py < height) {
                    block.add(Color.green(pixels[py * width + px]))
                }
            }
        }
        return block.toIntArray()
    }

    private fun sumAbsoluteDifference(block1: IntArray, block2: IntArray): Float {
        var sum = 0
        val count = minOf(block1.size, block2.size)
        for (i in 0 until count) {
            sum += kotlin.math.abs(block1[i] - block2[i])
        }
        return sum.toFloat()
    }

    private fun detectFlowAnomalies(
        vectors: List<FlowVector>,
        avgMagnitude: Float
    ): List<Pair<Float, Float>> {
        val anomalies = mutableListOf<Pair<Float, Float>>()
        val threshold = avgMagnitude * 2.5f

        for (vector in vectors) {
            if (vector.magnitude > threshold && threshold > 1f) {
                anomalies.add(
                    Pair(vector.startX, vector.startY)
                )
            }
        }

        return anomalies
    }
}
```

```kotlin
            start()
        }

        startTime = System.currentTimeMillis()

    }

                    filePath = file.absolutePath,
                    durationMs = duration,
                    fileSizeBytes = file.length()
                )
            } else null
        } catch (e: Exception) {
            null
        }
    }

            System.currentTimeMillis() - startTime

    )
```

```kotlin
    fun stitchPanorama(
        frames: List<Bitmap>,
        outputFileName: String = "panorama_${System.currentTimeMillis()}.jpg"
    ): Bitmap? {
        if (frames.size < 2) return frames.firstOrNull()

        val totalWidth = frames.sumOf { it.width }
        val maxHeight = frames.maxOf { it.height }

        val panorama = Bitmap.createBitmap(totalWidth, maxHeight, Bitmap.Config.ARGB_8888)
        val canvas = Canvas(panorama)
        val paint = Paint(Paint.ANTI_ALIAS_FLAG)

        var offsetX = 0
        for (frame in frames) {
            canvas.drawBitmap(frame, offsetX.toFloat(), 0f, paint)
            offsetX += frame.width
        }

        return panorama
    }

    fun stitchComparisonVideo(
        beforeFrames: List<Bitmap>,
        afterFrames: List<Bitmap>,
        outputDir: File
    ): StitchingResult? {
        val frameCount = maxOf(beforeFrames.size, afterFrames.size)
        val frameWidth = 1920
        val frameHeight = 540

        val halfWidth = frameWidth / 2
        val scaledBeforeFrames = beforeFrames.map {
            Bitmap.createScaledBitmap(it, halfWidth, frameHeight, true)
        }
        val scaledAfterFrames = afterFrames.map {
            Bitmap.createScaledBitmap(it, halfWidth, frameHeight, true)
        }

        val outputFile = File(outputDir, "comparison_${System.currentTimeMillis()}.mp4")
        val combinedFrames = mutableListOf<Bitmap>()

        for (i in 0 until frameCount) {
            val beforeFrame = scaledBeforeFrames.getOrElse(i) { scaledBeforeFrames.last() }
            val afterFrame = scaledAfterFrames.getOrElse(i) { scaledAfterFrames.last() }

            val combined = Bitmap.createBitmap(frameWidth, frameHeight, Bitmap.Config.ARGB_8888)
            val canvas = Canvas(combined)
            val paint = Paint(Paint.ANTI_ALIAS_FLAG)

            canvas.drawBitmap(beforeFrame, 0f, 0f, paint)
            canvas.drawBitmap(afterFrame, halfWidth.toFloat(), 0f, paint)

            canvas.drawLine(
                halfWidth.toFloat(), 0f,
                halfWidth.toFloat(), frameHeight.toFloat(),
                Paint().apply {
                    color = android.graphics.Color.WHITE
                    strokeWidth = 4f
                }
            )

            combinedFrames.add(combined)
        }

        return StitchingResult(
            outputPath = outputFile.absolutePath,
            frameCount = combinedFrames.size,
            durationMs = (frameCount * 33).toLong()
        )
    }

    fun createTimelapse(
        frames: List<Bitmap>,
        outputDir: File,
        fps: Int = 30
    ): File {
        val outputFile = File(outputDir, "timelapse_${System.currentTimeMillis()}.mp4")
```

```kotlin
                    onProgress = { progress = it }
                )
                analysisResult = result
            } catch (e: Exception) {
                error = "فشل التحليل: ${e.message}"
            } finally {
                isAnalyzing = false
            }
        }
    }

    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("تحليل الفيديو") },
                navigationIcon = {
                    IconButton(onClick = onNavigateBack) {
                        Icon(Icons.Default.ArrowBack, "رجوع")
                    }
                }
            )
        }
    ) { padding ->
        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding)
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            if (!isAnalyzing && analysisResult == null && error == null) {
                item {
                    Card {
                        Column(
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(24.dp),
                            horizontalAlignment = Alignment.CenterHorizontally
                        ) {
                            Icon(
                                Icons.Default.VideoFile,
                                null,
                                modifier = Modifier.size(64.dp),
                                tint = MaterialTheme.colorScheme.primary
                            )
                            Spacer(Modifier.height(16.dp))
                            Text("الفيديو جاهز للتحليل", fontWeight = FontWeight.Bold)
                            Spacer(Modifier.height(8.dp))
                            Text(
                                "سيتم تحليل الفيديو إطاراً بإطار لاكتشاف الأضرار والحركة",
                                style = MaterialTheme.typography.bodyMedium
                            )
                            Spacer(Modifier.height(16.dp))
                            Button(
                                onClick = { startAnalysis() },
                                modifier = Modifier.fillMaxWidth()
                            ) {
                                Icon(Icons.Default.Analytics, null)
                                Spacer(Modifier.width(8.dp))
                                Text("بدء التحليل")
                            }
                        }
                    }
                }
            }

            if (isAnalyzing) {
                item {
                    Card {
                        Column(
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(24.dp),
                            horizontalAlignment = Alignment.CenterHorizontally
                        ) {
                            CircularProgressIndicator(
                                progress = { progress },
                                modifier = Modifier.size(64.dp)
                            )
                            Spacer(Modifier.height(16.dp))
                            Text("جاري تحليل الفيديو...")
                            Text(
                                "${(progress * 100).toInt()}%",
                                style = MaterialTheme.typography.headlineMedium,
                                color = MaterialTheme.colorScheme.primary
                            )
                        }
                    }
                }
            }

            error?.let { msg ->
                item {
                    Card(colors = CardDefaults.cardColors(
                        containerColor = MaterialTheme.colorScheme.errorContainer
                    )) {
                        Row(Modifier.padding(16.dp)) {
                            Icon(Icons.Default.Error, null)
                            Spacer(Modifier.width(8.dp))
                            Text(msg)
                        }
                    }
                }
            }

            analysisResult?.let { result ->
                item {
                    Card {
                        Column(Modifier.padding(16.dp)) {
                            Text("ملخص التحليل", fontWeight = FontWeight.Bold)
                            Spacer(Modifier.height(8.dp))
                            Text("الإطارات الكلية: ${result.totalFrames}")
                            Text("الإطارات المحللة: ${result.analyzedFrames}")
                            Text("إطارات بها ضرر: ${result.framesWithDamage}")
                            Text("التقييم العام: ${result.overallAssessment}")
                            Text("المدة: ${result.durationMs / 1000} ثانية")
                        }
                    }
                }

                result.surfaceQuality?.let { quality ->
                    item {
                        Card(colors = CardDefaults.cardColors(
                            containerColor = MaterialTheme.colorScheme.secondaryContainer
                        )) {
                            Column(Modifier.padding(16.dp)) {
                                Text("جودة السطح", fontWeight = FontWeight.Bold)
                                Text("الدرجة: ${"%.1f".format(quality.overallScore)}%")
                                Text("العيوب: ${quality.defectCount}")
                            }
                        }
                    }
                }

                result.alignmentIssues?.let { issues ->
                    item {
                        Text("مشاكل المحاذاة (${issues.size})", fontWeight = FontWeight.Bold)
                    }
                    items(issues) { issue ->
                        Card {
                            Row(Modifier.padding(12.dp)) {
                                Icon(
                                    if (issue.isAligned) Icons.Default.CheckCircle
                                    else Icons.Default.Warning,
                                    null,
                                    tint = if (issue.isAligned) MaterialTheme.colorScheme.tertiary
                                    else MaterialTheme.colorScheme.error
                                )
                                Spacer(Modifier.width(8.dp))
                                Column {
                                    Text(issue.partName, fontWeight = FontWeight.Bold)
                                    Text(issue.description)
                                    Text("عرض الفجوة: ${"%.1f".format(issue.gapWidth)} مم")
                                }
                            }
                        }
                    }
                }

                if (result.framesWithDamage > 0) {
                    item {
                        Text("إطارات الضرر", fontWeight = FontWeight.Bold)
                    }
                    items(result.damageReports.filter { it.damageReport.regions.isNotEmpty() }.take(20)) { frame ->
                        FrameDamageCard(frame)
                    }
                }
            }
        }
    }
}

@Composable
fun FrameDamageCard(frame: FrameDamageReport) {
    Card {
        Row(Modifier.padding(12.dp)) {
            Column(Modifier.weight(1f)) {
                Text("إطار ${frame.frameNumber}", fontWeight = FontWeight.Bold)
                Text("الوقت: ${frame.timestampMs / 1000} ثانية")
                Text("مناطق الضرر: ${frame.damageReport.regions.size}")
            }
            Text(
                frame.damageReport.estimatedSeverity,
                color = when (frame.damageReport.estimatedSeverity) {
                    "HIGH" -> MaterialTheme.colorScheme.error
                    "MEDIUM" -> MaterialTheme.colorScheme.tertiary
                    else -> MaterialTheme.colorScheme.primary
                }
            )
        }
    }
}
```
