            if (damageAnalysis != null) {

            if (runAttemptCount < 3) {
                Result.retry()
            } else {
                Result.failure()
            }
        }

    private suspend fun performCloudOcr(bitmap: Bitmap): Map<String, Any>? {
        return try {
            val scaledBitmap = Bitmap.createScaledBitmap(bitmap, 640, 480, true)
            val outputStream = ByteArrayOutputStream()
            scaledBitmap.compress(Bitmap.CompressFormat.JPEG, 80, outputStream)
            val imageBytes = outputStream.toByteArray()

            val url = URL("https://vision.googleapis.com/v1/images:annotate")
            val connection = url.openConnection() as HttpURLConnection
            connection.requestMethod = "POST"
            connection.setRequestProperty("Content-Type", "application/json")
            connection.doOutput = true

            val requestBody = gson.toJson(mapOf(
                "requests" to listOf(
                    mapOf(
                        "image" to mapOf("content" to android.util.Base64.encodeToString(imageBytes, android.util.Base64.DEFAULT)),
                        "features" to listOf(mapOf("type" to "TEXT_DETECTION"))
                    )
                )
            ))

            connection.outputStream.write(requestBody.toByteArray())

            if (connection.responseCode == 200) {
                val response = connection.inputStream.bufferedReader().readText()
                val responseMap = gson.fromJson(response, Map::class.java)
                val responses = responseMap["responses"] as? List<Map<String, Any>>
                val textAnnotations = responses?.firstOrNull()?.get("textAnnotations") as? List<Map<String, Any>>
                val detectedText = textAnnotations?.firstOrNull()?.get("description") as? String ?: ""

                mapOf(
                    "detectedText" to detectedText,
                    "source" to "google_vision"
                )
            } else null
        } catch (e: Exception) {
            null
        }
    }

    private suspend fun performVehicleRecognition(bitmap: Bitmap): Map<String, Any>? {
        return try {

package com.workshoptech.workers

import android.content.Context
import androidx.work.*
import com.workshoptech.data.AppDatabase
import java.io.File
import java.io.FileInputStream
import java.io.FileOutputStream
import java.util.concurrent.TimeUnit
import java.util.zip.ZipEntry
import java.util.zip.ZipOutputStream

class SyncWorker(
    context: Context,
    params: WorkerParameters
) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result {
        return try {
            val backupFile = createBackup()
            Result.success(
                Data.Builder()
                    .putString("backupPath", backupFile.absolutePath)
                    .build()
            )
        } catch (e: Exception) {
            Result.retry()
        }
    }

    private fun createBackup(): File {
        val backupDir = File(applicationContext.filesDir, "backups")
        if (!backupDir.exists()) backupDir.mkdirs()

        val timestamp = java.text.SimpleDateFormat("yyyyMMdd_HHmmss", java.util.Locale.US).format(java.util.Date())
        val zipFile = File(backupDir, "backup_$timestamp.zip")

        ZipOutputStream(FileOutputStream(zipFile)).use { zos ->
            val dbFile = applicationContext.getDatabasePath("workshop_tech.db")
            if (dbFile.exists()) {
                zos.putNextEntry(ZipEntry("database/workshop_tech.db"))
                FileInputStream(dbFile).copyTo(zos)
                zos.closeEntry()
            }

            val filesDir = applicationContext.filesDir
            filesDir.walkTopDown().forEach { file ->
                if (file.isFile && file.extension != "zip") {
                    val relativePath = file.relativeTo(filesDir).path
                    zos.putNextEntry(ZipEntry("files/$relativePath"))
                    FileInputStream(file).copyTo(zos)
                    zos.closeEntry()
                }
            }
        }

        return zipFile
    }

    companion object {
        fun schedule(context: Context) {
            val constraints = Constraints.Builder()
                .setRequiredNetworkType(NetworkType.CONNECTED)
                .setRequiresCharging(true)
                .build()

            val request = PeriodicWorkRequestBuilder<SyncWorker>(
                24, TimeUnit.HOURS
            )
                .setConstraints(constraints)
                .addTag("daily_sync")
                .build()

            WorkManager.getInstance(context).enqueueUniquePeriodicWork(
                "daily_sync",
                ExistingPeriodicWorkPolicy.KEEP,
                request
            )
        }
    }
}

package com.workshoptech.util

import android.content.Context
import android.graphics.*
import android.net.Uri
import androidx.exifinterface.media.ExifInterface
import java.io.File
import java.io.FileOutputStream
import java.io.IOException

object ImageUtils {

    fun compressImage(
        context: Context,
        sourceUri: Uri,
        maxWidth: Int = 1024,
        maxHeight: Int = 1024,
        quality: Int = 85
    ): File? {
        return try {
            val inputStream = context.contentResolver.openInputStream(sourceUri)
            val originalBitmap = BitmapFactory.decodeStream(inputStream)
            inputStream?.close()

            if (originalBitmap == null) return null

            val ratio = minOf(
                maxWidth.toFloat() / originalBitmap.width,
                maxHeight.toFloat() / originalBitmap.height,
                1f
            )

            val newWidth = (originalBitmap.width * ratio).toInt()
            val newHeight = (originalBitmap.height * ratio).toInt()

            val scaledBitmap = Bitmap.createScaledBitmap(originalBitmap, newWidth, newHeight, true)
            originalBitmap.recycle()

            val compressedFile = File(context.cacheDir, "compressed_${System.currentTimeMillis()}.jpg")
            FileOutputStream(compressedFile).use { outputStream ->
                scaledBitmap.compress(Bitmap.CompressFormat.JPEG, quality, outputStream)
            }
            scaledBitmap.recycle()

            compressedFile
        } catch (e: IOException) {
            null
        }
    }

    fun createThumbnail(
        sourceFile: File,
        maxSize: Int = 256
    ): File? {
        return try {
            val originalBitmap = BitmapFactory.decodeFile(sourceFile.absolutePath)
            if (originalBitmap == null) return null

            val ratio = minOf(
                maxSize.toFloat() / originalBitmap.width,
                maxSize.toFloat() / originalBitmap.height
            )

            val newWidth = (originalBitmap.width * ratio).toInt()
            val newHeight = (originalBitmap.height * ratio).toInt()

            val thumbnail = Bitmap.createScaledBitmap(originalBitmap, newWidth, newHeight, true)
            originalBitmap.recycle()

            val thumbnailFile = File(sourceFile.parent, "thumb_${sourceFile.name}")
            FileOutputStream(thumbnailFile).use { outputStream ->
                thumbnail.compress(Bitmap.CompressFormat.JPEG, 70, outputStream)
            }
            thumbnail.recycle()

            thumbnailFile
        } catch (e: IOException) {
            null
        }
    }

    fun correctImageRotation(context: Context, imageUri: Uri): Bitmap? {
        return try {
            val inputStream = context.contentResolver.openInputStream(imageUri)
            val bitmap = BitmapFactory.decodeStream(inputStream)
            inputStream?.close()

            val exifInterface = imageUri.path?.let { ExifInterface(it) }
            val orientation = exifInterface?.getAttributeInt(
                ExifInterface.TAG_ORIENTATION,
                ExifInterface.ORIENTATION_NORMAL
            )

            val rotation = when (orientation) {
                ExifInterface.ORIENTATION_ROTATE_90 -> 90f
                ExifInterface.ORIENTATION_ROTATE_180 -> 180f
                ExifInterface.ORIENTATION_ROTATE_270 -> 270f
                else -> 0f
            }

            if (rotation != 0f && bitmap != null) {
                val matrix = Matrix()
                matrix.postRotate(rotation)
                val rotated = Bitmap.createBitmap(bitmap, 0, 0, bitmap.width, bitmap.height, matrix, true)
                bitmap.recycle()
                rotated
            } else {
                bitmap
            }
        } catch (e: IOException) {
            null
        }
    }

    fun cropImage(bitmap: Bitmap, rect: Rect): Bitmap {
        return Bitmap.createBitmap(
            bitmap,
            rect.left.coerceIn(0, bitmap.width),
            rect.top.coerceIn(0, bitmap.height),
            rect.width().coerceAtMost(bitmap.width - rect.left),
            rect.height().coerceAtMost(bitmap.height - rect.top)
        )
    }
}

        return calendar.timeInMillis
    }

    fun getStartOfMonth(): Long {
        val calendar = Calendar.getInstance()
        calendar.set(Calendar.DAY_OF_MONTH, 1)
        calendar.set(Calendar.HOUR_OF_DAY, 0)
        calendar.set(Calendar.MINUTE, 0)
        calendar.set(Calendar.SECOND, 0)
        return calendar.timeInMillis
    }

    fun daysBetween(start: Long, end: Long): Long {
        return TimeUnit.MILLISECONDS.toDays(end - start)
    }

    fun hoursBetween(start: Long, end: Long): Long {
        return TimeUnit.MILLISECONDS.toHours(end - start)
    }

    fun isToday(timestamp: Long): Boolean {
        val cal1 = Calendar.getInstance()
        val cal2 = Calendar.getInstance()
        cal1.timeInMillis = timestamp
        cal2.timeInMillis = System.currentTimeMillis()
        return cal1.get(Calendar.YEAR) == cal2.get(Calendar.YEAR) &&
               cal1.get(Calendar.DAY_OF_YEAR) == cal2.get(Calendar.DAY_OF_YEAR)
    }

    fun isYesterday(timestamp: Long): Boolean {
        val cal1 = Calendar.getInstance()
        val cal2 = Calendar.getInstance()
        cal1.timeInMillis = timestamp
        cal2.timeInMillis = System.currentTimeMillis()
        cal2.add(Calendar.DAY_OF_YEAR, -1)
        return cal1.get(Calendar.YEAR) == cal2.get(Calendar.YEAR) &&
               cal1.get(Calendar.DAY_OF_YEAR) == cal2.get(Calendar.DAY_OF_YEAR)
    }

    fun getRelativeTime(timestamp: Long): String {
        val now = System.currentTimeMillis()
        val diff = now - timestamp

        return when {
            diff < TimeUnit.MINUTES.toMillis(1) -> "الآن"
            diff < TimeUnit.HOURS.toMillis(1) -> "منذ ${TimeUnit.MILLISECONDS.toMinutes(diff)} دقيقة"
            diff < TimeUnit.DAYS.toMillis(1) -> "منذ ${TimeUnit.MILLISECONDS.toHours(diff)} ساعة"
            isYesterday(timestamp) -> "أمس"
            diff < TimeUnit.DAYS.toMillis(7) -> "منذ ${TimeUnit.MILLISECONDS.toDays(diff)} أيام"
            diff < TimeUnit.DAYS.toMillis(30) -> "منذ ${TimeUnit.MILLISECONDS.toDays(diff) / 7} أسابيع"
            else -> formatArabicDate(timestamp)
        }
    }

object CurrencyFormatter {

        "LYD" to "د.ل",
        "EGP" to "ج.م",
        "SAR" to "ر.س",
        "AED" to "د.إ",
        "KWD" to "د.ك",
        "QAR" to "ر.ق",
        "BHD" to "د.ب",
        "OMR" to "ر.ع",
        "JOD" to "د.أ",
        "LBP" to "ل.ل",
        "SYP" to "ل.س",
        "IQD" to "د.ع",
        "YER" to "ر.ي",
        "SDP" to "ج.س",
        "TND" to "د.ت",
        "DZD" to "د.ج",
        "MAD" to "د.م",
        "MRU" to "أ.م",
        "SOS" to "ش.ص",
        "DJF" to "ف.ج",
        "KMF" to "ف.ق"
    )

    fun format(amount: Double, currencyCode: String): String {
        val symbol = currencySymbols[currencyCode] ?: currencyCode
        val formatter = NumberFormat.getNumberInstance(Locale("ar"))
        formatter.maximumFractionDigits = 2
        formatter.minimumFractionDigits = 2
        return "${formatter.format(amount)} $symbol"
    }

    fun formatShort(amount: Double, currencyCode: String): String {
        val symbol = currencySymbols[currencyCode] ?: currencyCode
        return when {
            amount >= 1_000_000 -> "${"%.1f".format(amount / 1_000_000)}M $symbol"
            amount >= 1_000 -> "${"%.1f".format(amount / 1_000)}K $symbol"
            else -> "${"%.0f".format(amount)} $symbol"
        }
    }

    fun getSymbol(currencyCode: String): String {
        return currencySymbols[currencyCode] ?: currencyCode
    }

object ValidationUtils {

    fun isValidPhone(phone: String, countryCode: String): Boolean {
        val cleaned = phone.replace(Regex("[^\\d+]"), "")
        return when (countryCode) {
            "LY" -> cleaned.matches(Regex("^\\+?218\\d{9}$")) || cleaned.matches(Regex("^0\\d{9}$"))
            "EG" -> cleaned.matches(Regex("^\\+?20\\d{10}$")) || cleaned.matches(Regex("^0\\d{10}$"))
            "SA" -> cleaned.matches(Regex("^\\+?966\\d{9}$")) || cleaned.matches(Regex("^0\\d{9}$"))
            "AE" -> cleaned.matches(Regex("^\\+?971\\d{9}$")) || cleaned.matches(Regex("^0\\d{9}$"))
            else -> cleaned.length >= 10
        }
    }

    fun isValidPlate(plate: String, countryId: String): Boolean {
        val patterns = com.workshoptech.domain.util.PlatePatterns.getForCountry(countryId)
        return patterns.any { pattern ->
            try {
                Regex(pattern).matches(plate.trim())
            } catch (e: Exception) {
                false
            }
        }
    }

    fun isValidColorCode(code: String): Boolean {
        return code.matches(Regex("^[A-Za-z0-9]{1,6}$"))
    }

    fun isValidVin(vin: String): Boolean {
        return vin.length == 17 && !vin.contains('I') && !vin.contains('O') && !vin.contains('Q')
    }

    fun sanitizeInput(input: String): String {
        return input
            .trim()
            .replace(Regex("[<>\"'%;()&+]"), "")
            .take(200)
    }
}
