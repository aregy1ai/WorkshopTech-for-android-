```kotlin
package com.workshoptech.data

import android.content.Context
import androidx.room.Database
import androidx.room.Room
import androidx.room.RoomDatabase
import com.workshoptech.data.dao.*
import com.workshoptech.data.entity.*

@Database(
    entities = [
        CaseEntity::class,
        CasePhotoEntity::class,
        CustomerEntity::class,
        DamageFindingEntity::class,
        InspectionEntity::class
    ],
    version = 1
)
abstract class AppDatabase : RoomDatabase() {
    abstract fun caseDao(): CaseDao
    abstract fun customerDao(): CustomerDao
    abstract fun photoDao(): PhotoDao
    abstract fun inspectionDao(): InspectionDao

    companion object {
        @Volatile
        private var INSTANCE: AppDatabase? = null

        fun getInstance(context: Context): AppDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    "workshop_tech.db"
                ).build()
                INSTANCE = instance
                instance
            }
        }
    }
}
```

17. data/repository/WorkshopRepository.kt

```kotlin
package com.workshoptech.data.repository

import com.workshoptech.data.dao.*
import com.workshoptech.data.entity.*
import kotlinx.coroutines.flow.Flow

class WorkshopRepository(
    private val caseDao: CaseDao,
    private val customerDao: CustomerDao,
    private val photoDao: PhotoDao,
    private val inspectionDao: InspectionDao
) {
    fun observeCases(query: String?): Flow<List<CaseEntity>> {
        return if (query.isNullOrBlank()) caseDao.observeAll()
        else caseDao.search(query)
    }

    fun observeCase(caseId: String): Flow<CaseEntity?> = caseDao.observeById(caseId)

    suspend fun upsertCase(case: CaseEntity) = caseDao.upsert(case)

    suspend fun updateCaseStatus(caseId: String, status: String) {
        caseDao.updateStatus(caseId, status, System.currentTimeMillis())
    }

    fun observeCustomers(query: String?): Flow<List<CustomerEntity>> {
        return if (query.isNullOrBlank()) customerDao.observeAll()
        else customerDao.search(query)
    }

    suspend fun findCustomerByPhone(phone: String): CustomerEntity? =
        customerDao.findByPhone(phone)

    suspend fun upsertCustomer(customer: CustomerEntity) = customerDao.upsert(customer)

    fun observePhotos(caseId: String): Flow<List<CasePhotoEntity>> =
        photoDao.observeByCase(caseId)

    suspend fun addPhoto(photo: CasePhotoEntity) = photoDao.insert(photo)

    fun observeInspections(caseId: String): Flow<List<InspectionEntity>> =
        inspectionDao.observeByCase(caseId)

    suspend fun upsertInspection(inspection: InspectionEntity) =
        inspectionDao.upsert(inspection)

    suspend fun getLatestInspection(caseId: String, type: String): InspectionEntity? =
        inspectionDao.getLatestByType(caseId, type)
}
```

22. ml/OcrEngine.kt

```kotlin
package com.workshoptech.ml

import android.graphics.Bitmap
import com.google.mlkit.vision.common.InputImage
import com.google.mlkit.vision.text.TextRecognition
import com.google.mlkit.vision.text.latin.TextRecognizerOptions
import com.workshoptech.domain.model.PlateResult
import com.workshoptech.domain.util.PlatePatterns
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.tasks.await
import kotlinx.coroutines.withContext

object OcrEngine {
    private val recognizer = TextRecognition.getClient(
        TextRecognizerOptions.DEFAULT_OPTIONS
    )

    suspend fun recognizePlate(
        bitmap: Bitmap,
        countryId: String
    ): PlateResult = withContext(Dispatchers.IO) {
        val image = InputImage.fromBitmap(bitmap, 0)
        val result = recognizer.process(image).await()
        val rawText = result.text ?: ""

        val cleaned = cleanText(rawText)
        val patterns = PlatePatterns.getForCountry(countryId)
        val matches = mutableListOf<String>()

        for (pattern in patterns) {
            try {
                val regex = Regex(pattern)
                if (regex.matches(cleaned)) {
                    matches.add(cleaned)
                }
            } catch (e: Exception) {
                // Skip invalid patterns
            }
        }

        val candidates = if (matches.isEmpty()) listOf(cleaned) else matches
        val confidence = if (matches.isNotEmpty()) 0.9f else 0.5f

        PlateResult(
            text = candidates.firstOrNull() ?: cleaned,
            rawText = rawText,
            confidence = confidence,
            candidates = candidates,
            country = countryId,
            needsReview = confidence < 0.7f
        )
    }

    private fun cleanText(text: String): String {
        return text
            .replace(" ", "")
            .replace("\n", "")
            .replace("٠", "0")
            .replace("١", "1")
            .replace("٢", "2")
            .replace("٣", "3")
            .replace("٤", "4")
            .replace("٥", "5")
            .replace("٦", "6")
            .replace("٧", "7")
            .replace("٨", "8")
            .replace("٩", "9")
            .uppercase()
            .trim()
    }
}
```

34. viewmodel/CreateCaseViewModel.kt

```kotlin
package com.workshoptech.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.workshoptech.data.entity.CaseEntity
import com.workshoptech.data.repository.WorkshopRepository
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import java.util.UUID

data class CreateCaseState(
    val country: String = "LY",
    val plateText: String = "",
    val plateTextRaw: String? = null,
    val makeId: String? = null,
    val modelId: String? = null,
    val year: Int? = null,
    val vehicleFreeText: String = "",
    val canSave: Boolean = false,
    val saving: Boolean = false
)

class CreateCaseViewModel(
    private val repository: WorkshopRepository
) : ViewModel() {
    private val _state = MutableStateFlow(CreateCaseState())
    val state: StateFlow<CreateCaseState> = _state.asStateFlow()

    fun setCountry(v: String) = update { it.copy(country = v) }
    fun setPlateText(v: String) = update { it.copy(plateText = v) }
    fun setMake(v: String?) = update { it.copy(makeId = v) }
    fun setModel(v: String?) = update { it.copy(modelId = v) }
    fun setYear(v: Int?) = update { it.copy(year = v) }
    fun setVehicleFreeText(v: String) = update { it.copy(vehicleFreeText = v) }

    fun saveCase(onSaved: (String) -> Unit) {
        val s = _state.value
        if (!s.canSave || s.saving) return

        viewModelScope.launch {
            update { it.copy(saving = true) }

            val caseId = UUID.randomUUID().toString()
            val now = System.currentTimeMillis()

            val entity = CaseEntity(
                caseId = caseId,
                plateCountry = s.country,
                plateText = s.plateText.trim(),
                plateTextRaw = s.plateTextRaw,
                customerId = null,
                vehicleMakeId = s.makeId,
                vehicleModelId = s.modelId,
                vehicleYear = s.year,
                vehicleFreeText = s.vehicleFreeText.ifBlank { null },
                colorCode = null,
                colorName = null,
                status = "NEW",
                createdAt = now,
                updatedAt = now,
                estimatedCost = null,
                actualCost = null,
                estimatedHours = null,
                actualHours = null
            )

            repository.upsertCase(entity)
            update { it.copy(saving = false) }
            onSaved(caseId)
        }
    }

    private fun update(block: (CreateCaseState) -> CreateCaseState) {
        val newState = block(_state.value)
        val plateOk = newState.plateText.trim().isNotEmpty()
        val vehicleOk = newState.vehicleFreeText.isNotBlank() ||
                (newState.makeId != null && newState.modelId != null)
        _state.value = newState.copy(canSave = plateOk && vehicleOk)
    }
}
```

35. viewmodel/CaseDetailViewModel.kt

```kotlin
package com.workshoptech.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.workshoptech.data.entity.CaseEntity
import com.workshoptech.data.entity.CasePhotoEntity
import com.workshoptech.data.repository.WorkshopRepository
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import java.util.UUID

data class CaseDetailState(
    val case: CaseEntity? = null,
    val photos: List<CasePhotoEntity> = emptyList()
)

class CaseDetailViewModel(
    private val repository: WorkshopRepository
) : ViewModel() {

    fun observeCase(caseId: String): StateFlow<CaseDetailState> {
        return combine(
            repository.observeCase(caseId),
            repository.observePhotos(caseId)
        ) { c, photos ->
            CaseDetailState(case = c, photos = photos)
        }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), CaseDetailState())
    }

    fun addPhoto(caseId: String, imagePath: String) {
        viewModelScope.launch {
            val photo = CasePhotoEntity(
                photoId = UUID.randomUUID().toString(),
                caseId = caseId,
                imagePath = imagePath,
                thumbnailPath = null,
                capturedAt = System.currentTimeMillis(),
                photoType = "GENERAL",
                ocrText = null,
                ocrConfidence = null,
                notes = null
            )
            repository.addPhoto(photo)
        }
    }
}
```
